dewata,202,106,4	script	Dewata Guide#dewata01	4_M_DEWWOMAN,{
	mes "[Dewata Guide]";
	mes "欢迎来到^8B4513德瓦塔岛^000000，";
	mes "这里海浪耀眼、景色迷人，是放松身心的好地方。";
	mes "有任何问题都可以问我。";
	next;
	mes "[Dewata Guide]";
	mes "要我带您去哪里？";
	F_Navi("[Dewata Guide]");
	while (1) {
		next;
		switch(select("[ 卡普拉职员 ]:[ 武器商人 ]:[ 防具商人 ]:[ 工具商人 ]:[ 餐馆老板 ]:[ 喀拉喀托看门人 ]:[ 艾尔贝塔水手 ]:移除小地图标记:取消")) {
		case 1:
			callsub L_Mark, "^0000FF卡普拉职员^000000";
			viewpoint 1,202,184,0,0x000FFF;
			break;
		case 2:
			callsub L_Mark, F_Navi("武器商人","dewata,218,164","^008080");
			viewpoint 1,218,163,1,0x008080;
			break;
		case 3:
			callsub L_Mark, F_Navi("防具商人","dewata,158,182","^008080");
			viewpoint 1,158,182,2,0x008080;
			break;
		case 4:
			callsub L_Mark, F_Navi("工具商人","dewata,182,164","^FF1493");
			viewpoint 1,182,164,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("餐馆老板","dewata,146,109","^FF1493");
			viewpoint 1,146,109,4,0xFF1493;
			break;
		case 6:
			callsub L_Mark, F_Navi("喀拉喀托看门人","dewata,235,56","^8A2BE2");
			viewpoint 1,235,56,5,0x8A2BE2;
			break;
		case 7:
			callsub L_Mark, F_Navi("艾尔贝塔水手","dewata,229,49","^00BFFF");
			viewpoint 1,229,49,6,0x00BFFF;
			break;
		case 8:
			mes "[Dewata Guide]";
			mes "我会移除您小地图上的所有标记。";
			mes "还有其他需要我帮忙的吗？";
			viewpoint 2,202,184,0,0xFFFFFF;
			viewpoint 2,218,163,1,0xFFFFFF;
			viewpoint 2,158,182,2,0xFFFFFF;
			viewpoint 2,182,164,3,0xFFFFFF;
			viewpoint 2,232,59,4,0xFFFFFF;
			viewpoint 2,195,235,5,0xFFFFFF;
			viewpoint 2,230,52,6,0xFFFFFF;
			break;
		case 9:
			mes "[Dewata Guide]";
			mes "祝您旅途愉快！！";
			close;
		}
	}

L_Mark:
	mes "[Dewata Guide]";
	mes getarg(0)+"已";
	mes "标记在您的小地图上。";
	mes "请查看小地图。";
	mes "还有其他需要我帮忙的吗？";
	return;
}
