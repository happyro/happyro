//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[Derek Shot]";
	mes "服装效果让你无法参加战斗吗？一定很烦人吧。";
	next;
	mes "[Derek Shot]";
	mes "虽然穿在你身上很好看，但也可能非常碍事。我只需挥挥手就能帮你去除它！！我会把一切整理好。准备好后请选择选项！";
	next;
	switch( select( "保留效果", "立即解除" ) ){
		case 1:
			mes "[Derek Shot]";
			mes "改变主意后再来找我。";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Derek Shot]";
			mes "解除已完成。现在应该可以自由行动了。";
			atcommand "@changedress";
			close;
	}
}
