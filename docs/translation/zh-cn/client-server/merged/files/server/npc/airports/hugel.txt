hugel,178,142,0	script	toairplane#hugel	45,1,1,{
	end;
OnTouch:
	mes "使用飞艇需要支付1,200 Zeny或一张免费飞艇票。";
	mes "要使用这项服务吗？";
	next;
	if (select("是:否") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane",244,58;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane",244,58;
			end;
		}
		mes "抱歉，你的钱不够。";
		mes "请记住，使用这项服务需要支付1,200 Zeny。";
		close;
	}
	mes "谢谢，欢迎下次再来。";
	close;
}
