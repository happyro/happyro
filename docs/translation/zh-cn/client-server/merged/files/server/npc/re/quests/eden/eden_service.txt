-	script	Goods Cabinet#00::pggc	-1,{
	mes "这是一个物品储藏柜。";
	mes "纸上写着一段留言。";
	next;
	mes "+ 储藏柜仅供 +";
	mes "+ 伊甸园集团使用 +";
	mes "如果你想使用这个";
	mes "储藏柜，请先检查";
	mes "储藏柜，请确认以下事项：";
	next;
	mes "1.你是伊甸园集团";
	mes "   的成员吗？";
	mes "2.你学会了";
	mes "   足够的基础技能吗？";
	mes "3.储藏柜费用为";
	mes "  ^4d4dff500 Zeny^000000！";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "使用储藏柜需要投入 Zeny。";
		mes "费用：500 Zeny";
		mes "要使用吗？";
		next;
		switch(select("使用储藏柜。:取消。")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "储藏柜无法为我工作。";
						mes "也许我还没有资格使用储藏柜。";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "我的 Zeny 不够。";
				mes "使用储藏柜至少需要 500 Zeny。";
				close;
			case 2:
				mes "下次我再使用。";
				close;
		}
	}
	mes "使用这个储藏柜需要伊甸园集团徽章。";
	close;
}
