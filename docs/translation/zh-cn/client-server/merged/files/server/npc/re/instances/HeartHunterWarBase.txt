ein_fild04,275,342,5	script	Striker Unit Commander#	4_M_REBELLION3,{
	if (terra_gloria_main < 19) {
		mes "[突击部队指挥官]";
		mes "嘿，谁在那里？";
		next;
		select("我正在和反叛军部队执行任务。");
		mes "[突击部队指挥官]";
		mes "这里很危险，你必须马上离开。我们就在敌方基地附近。";
		close;
	}
	switch( checkquest(12375,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[突击部队指挥官]";
		mes "我们正在整理基地内部情况。现在还不能让你进去。";
		close;
	case 2:
		mes "^0000ff基地内部似乎已经整理完毕。^000000";
		erasequest 12375;// Base maintenance
		break;
	}
	if (isbegin_quest(12374) == 0) {
		mes "[突击部队指挥官]";
		mes "行动准备好后，你和我一起进去。";
		next;
		mes "[突击部队指挥官]";
		mes "对了，你收到朱利安将军的简报了吗？收到后再回来。";
		close;
	}
	mes "[突击部队指挥官]";
	mes "行动准备好后，你和我一起进去。";
	next;
	if (terra_gloria_main == 19)
		.@md_name$ = "Heart Hunter War Base 1";
	else
		.@md_name$ = "Heart Hunter War Base 2";
	switch( instance_enter(.@md_name$) ) {
	case IE_NOMEMBER:
		mes "^0000ff没有所属部队（队伍）的冒险者似乎无法进入。^000000";
		close;
	case IE_NOINSTANCE:
		mes "[突击部队指挥官]";
		mes "行动区域是" + .@md_name$ + " Check whether the operation was applied properly.";
		next;
		mes "^ff0000未清理心灵猎手战争基地前禁止前往外围边界，也无法进入已经清理过的基地。请确认行动地点。^000000";
		next;
		mes "^ff0000你的进度与队长或队员不同时会出现此情况。请找一名能和你前往同一地点的队员。^000000";
		next;
		mes "^ff0000若未将纪念地下城创建为行动区域，也无法进入。^000000";
		close;
	case IE_OTHER:
		mes "发生未知错误。";
		close;
	case IE_OK:
		mapannounce "ein_fild04", "队伍中的" + strcharinfo(0) + "正在进入" + .@md_name$ + "。", bc_map, 0xFF99;
		setquest 12375;// Base maintenance
		// if (terra_gloria_main == 20)
		// 	warp "1@swat",35,65;
		// else
		// 	warp "1@swat",73,118;
		end;
	}
}

			mes "[Julian]";

			mes "[Julian]";

			next;
			mes "[Julian]";
			mes "总之，行动大多必须针对人员，并尽可能维护基地设施。";
			next;
			mes "[Julian]";
			mes "我们的先遣队已经确保了南部基地的边界地面。我们将沿那条路线前往中央区域，各突击小队则分别行动。";
			next;
			mes "[Julian]";
			mes "你们就像外籍军团，行动时待在我身边。还有问题吗？当然，即使有问题我也不会回答。";
		}
		next;
		setquest 12374;// Heart Hunter Base occupation
		mes "[Julian]";
		mes "之后听从你们小队长的指示。基地大门打开后，去和那边的突击小队指挥官交谈并开始行动。";

			next;
			mes "[Julian]";
			mes "如果你准备好了，我会通知你攻击时间。";
			next;
			if (select( "我准备好了。", "我还需要一点时间。" ) == 2) {
				mes "[Julian]";
				mes "明白了。我们时间紧迫，尽快准备好后告诉我。";
				close;
			}
			break;
		}
		close;
	case 1:
		if (is_party_leader() == false) {
			mes "[Julian]";
			mes "之后听从你们小队长的指示。基地大门打开后，去和那边的突击小队指挥官交谈并开始行动。";
			close;

			mes "[Julian]";
			mes "明白了。是我要求太多。";
			close;
		}
		break;
	}

	else
		instance_create("Heart Hunter War Base 2");

	mes "很快就会开放。基地大门打开后，去和那边的突击小队指挥官交谈并开始行动。";
	close;
