	case 1:

			mes "队伍名称：" + getpartyname( getcharid(1) );
			mes "队长：" + strcharinfo(0);
			mes "^0000ff" + .@md_name$ + " ^000000- 创建失败！";
			close3;
		}

		mes "^ff0000正在准备次元传送门……^000000";
		mes "^ff0000请稍候。^000000";
		close3;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			close3;
		case IE_NOINSTANCE:
			if (is_party_leader() == false) {
				mes "[Runane]";
				mes "正在传送……";
				next;
				mes "[Runane]";

				next;
			}
			mes "[Runane]";
			mes "^ff0000你的队长还没有创建时间裂缝。^000000";
			close3;
		case IE_OTHER:
			mes "发生未知错误。";
			close3;
		case IE_OK:
			setquest 13182;// Nightmarish Jitterbug: Waiting
			mapannounce "moc_para01", "" + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering the " + .@md_name$ + ".", bc_map, 0xFF99;

			specialeffect2 EF_PORTAL2;
			// warp "1@jtb",16,17;
			end;
		}
	case 3:
		cutin "",255;
		end;
	}
}
