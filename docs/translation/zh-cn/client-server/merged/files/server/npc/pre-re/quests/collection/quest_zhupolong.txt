lou_dun02,161,264,4	script	Dragon Hunter	732,{

	if (checkquest(60182) != -1) {
		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[猎龙人]";
			mes "你已经猎杀了 50 条朱波隆吗？";
			next;
			switch(select("是:否:我想退出")) {
			case 1:
				mes "[猎龙人]";
				mes "嗯，我不认为你已经猎杀了 50 条。";
				close;
			case 2:
				mes "[猎龙人]";
				mes "记住，我需要你帮忙猎杀朱波隆，所以去猎杀 50 条吧。";
				close;
			case 3:
				mes "[猎龙人]";
				mes "确定要停止狩猎吗？";
				mes "你取得的所有进度都会被清除";
				next;
				switch(select("是:否")) {
				case 1:
					mes "[猎龙人]";
					mes "好的，改变主意后再回来。";
					erasequest 60182;
					close;
				case 2:
					mes "[猎龙人]";
					mes "请猎杀 50 条朱波隆，让这里成为大家获取经验的地方。";
					close;
				}
			}
		}
		mes "[猎龙人]";
		mes "太棒了，你完成得真快。";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[猎龙人]";
		mes "太好了！";
		mes "你是来帮我的，对吧？";
		mes "不然我真不知道该拿这里的怪物怎么办！";
		next;
		switch(select("当然:不了")) {
		case 1:
			mes "[猎龙人]";
			mes "太好了！我一看到你就知道可以依靠你！";
			mes "我需要你减少目前在这张地图上肆虐的怪物数量！";
			mes "这里的朱波隆太多了，低等级玩家无法获得良好经验。";
			next;
			mes "[猎龙人]";
			mes "那么，你还愿意帮我吗？";
			next;
			switch(select("好的:取消")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[猎龙人]";
					mes "感谢你为朱波隆提供的帮助！";
					mes "现在情况稳定了，我不再需要你帮忙猎杀它们。";
					close;
				}
				mes "[猎龙人]";
				mes "猎杀 50 条朱波隆，完成后告诉我。我会奖励你的努力。";
				setquest 60182;
				close;
			case 2:
				mes "[猎龙人]";
				mes "改变主意后请再回来。";
				close;
			}
		case 2:
			mes "[猎龙人]";
			mes "改变主意后请再回来。";
			close;
		}
	}
	mes "[猎龙人]";
	mes "哎呀，哎呀！";
	next;
	mes "[猎龙人]";
	mes "抱歉我现在无法交谈，这里痛苦不堪，这些该死的朱波隆快要杀了我。";
	close;
}
