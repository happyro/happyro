//===== Description: =========================================
//= Plays a hybrid Russian Roulette Rock Scissors Paper game.
//===== Additional Comments: =================================
//= Prizes customizable, Added emotions.
//= 1.2 Fixes by Blackthunder and me [Poki#3]

cmd_in02,182,126,2	script	Crazy Boris	85,{
	set .@counter,1;
	mes "嘿，想玩猜拳俄罗斯轮盘吗？";
	next;
	menu "让我玩。",PLAY,"说明规则。",RULES,"离开",LEAVE;
SAME:
	mes "平局！再来一次！";
	next;
PLAY:
	mes "石头……布……";
	set .@opp, rand(1,3);
	menu "^0000FF石头！",-,"^FF0000剪刀！",SCISSORS,"^00FF00布！^000000",PAPER;

  	if (.@lastchoice == 1) set .@opp, rand(1,3);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,1;
	if (.@opp == 1) goto SAME;
	if (.@opp == 2) goto WIN;
	goto LOSE;

SCISSORS:
	if (.@lastchoice == 2) set .@opp,rand(1,2);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,2;
	if (.@opp == 1) goto LOSE;
	if (.@opp == 2) goto SAME;
	goto WIN;

PAPER:
	if (.@lastchoice == 3) set .@opp,rand(2,3);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,3;
	if (.@opp == 1) goto WIN;
	if (.@opp == 2) goto LOSE;
	goto SAME;

WIN:
	mes "可恶，您赢了！";
	emotion ET_PROFUSELY_SWEAT;
	next;
	set .@win, 1;
	goto YOUPULL;

LOSE:
	emotion ET_SMILE;
	mes "哇哈哈！您输了！";
	next;
	set .@win, 0;

YOUPULL:
	mes .@counter +" of 6";
	if ( .@counter == 6 )
		mes "准备祈祷吧。";
	set .@pull, rand( 1,( 7 -.@counter ) );
	set .@counter, .@counter +1;
	next;
	if ( .@pull == 1 ) {
		if ( .@win ) goto KILL;
		specialeffect2 EF_SUI_EXPLOSION;
		emotion ET_KIK;
		percentheal -100,-100;
		mes "*^0000FF咔哒^000000* *^FF0000砰^000000*";
		mes "您死了！";
		close;
	}
	emotion ET_HNG;
	mes "*^0000FF咔哒^000000* 呼……";
	goto PLAY;

RULES:
	mes "好了，规则如下：";
	mes "我有一把 ^FF00006^000000 发膛的手枪，里面装着 ^FF00001^000000 发子弹。我们先玩^FF0000剪刀 ^00FF00布 ^0000FF石头^000000。输家扣动扳机，最后活下来的人获胜。";
	mes "赢过我就能获得奖品。";
	next;
	menu "让我玩。",-,"不用了，谢谢。",LEAVE;
	mes "好了，开始吧……";
	next;
	goto PLAY;

KILL:
	specialeffect EF_SUI_EXPLOSION;
	emotion ET_HUK;
	mes "*^0000FFClick^000000* *^FF0000BANG^000000*";
	mes "哎哟 @#$%^！！痛死我了！！";
	next;
	mes "恭喜！您赢得了……";
	switch( rand( 1,10 ) ) {
		case 1: setarray .@reward[0], 10,984; break;
		case 3: setarray .@reward[0],100,601; break;	// 100x Fly Wings
		case 4: setarray .@reward[0],  8,603; break;	// 8x Old Blue Box
		case 5: setarray .@reward[0],  4,617; break;	// 4x Old Violet Box
		case 6: setarray .@reward[0],  1,616; break;	// 1x Old Card Album
		case 7: setarray .@reward[0], 10,604; break;	// 10x Dead Branch
		case 8: setarray .@reward[0],  3,969; break;	// 3x Gold
		case 10: setarray .@reward[0],20,505; break;	// 20x Blue Potion
		case 2:
		case 9:
			setarray .@reward[0], 10,985;
	}
	mes .@reward[0] +" 个"+ getitemname( .@reward[1] ) +"！";
	getitem .@reward[1], .@reward[0];
	close;

LEAVE:
	mes "胆小鬼。";
	close;
}
