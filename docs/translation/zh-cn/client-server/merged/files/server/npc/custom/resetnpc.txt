//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prontera,150,193,4	script	Reset Girl	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[Reset Girl]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "抱歉，您一生只能重置 "+callfunc("F_InsertPlural",.@Reset[3],"次")+"。";
		close;
	}
	mes "我是重置少女。";
	mes "重置属性："+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	mes "重置技能："+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	mes "全部重置："+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "您只能重置 "+callfunc("F_InsertPlural",.@Reset[3],"次")+"，请谨慎使用。";
	mes "请选择要使用的服务：";
	next;
	set .@i,(select("^FF3355重置技能:重置属性:全部重置^000000:取消"));
	if(.@i > 3) close;
	mes "[Reset Girl]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "抱歉，您的 Zeny 不足。";
		close;
	}
	if(.@Reset[3]){
		mes "您一生只能重置 "+callfunc("F_InsertPlural",.@Reset[3],"次")+"，确定要继续吗？";
		if(select("让我想想:没问题") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "完成！";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
}
