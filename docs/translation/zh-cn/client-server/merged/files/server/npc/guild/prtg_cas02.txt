-	script	Swanhild#1::OutsideFlagsPR2	722,{

	set .@GID, GetCastleData("prtg_cas02",1);

	if (.@GID == 0) {
		mes "[ 神圣卢恩-米德加尔特王国法令 ]";
		mes " ";
		mes "1. 遵循神圣卢恩-米德加尔特王国法令，";
		mes "我们宣布";
		mes "这座城堡目前没有正式领主。";
		mes " ";
		mes "2. 对于能够";
		mes "克服所有试炼";
		mes "并摧毁城堡核心的人，";
		mes "国王将授予其";
		mes "这座城堡的所有权。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 回声 ]";
		mes "勇士们……";
		mes "您想回到荣耀之地吗？";
		next;
		if(select("返回公会城堡:退出") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("prtg_cas02",1)) {
				warp "prtg_cas02",169,55;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神圣卢恩-米德加尔特王国法令 ]";
	mes " ";
	mes "1. 遵循神圣卢恩-米德加尔特王国法令，";
	mes "我们批准此地归";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公会私人所有。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公会的会长是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如有人对此有异议，";
	mes "请用手中的钢刃证明您的力量与荣誉。";
	close;

OnRecvCastlePt02:
	FlagEmblem GetCastleData("prtg_cas02",1);
	end;
}
