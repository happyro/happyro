gef_fild10,242,202,0	script	Dimensional Gorge Piece	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Orc's Memory";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "只有等级 ^ff000030 ~ 80^000000 的玩家才能进入此副本。";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "队伍状态已确认。要预约进入 "+.@md_name$+" 吗？";
			next;
			switch(select("预约 "+.@md_name$+":进入副本:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "队伍名称："+getpartyname(.@party_id);
					mes "队长："+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000 - 预约失败。";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000- 正在预约入口";
				mes "预约后想进入 "+.@md_name$+"，必须从菜单选择‘进入副本’。";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + " 进入纪念副本:取消") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "副本创建后即可进入。";
		next;
		if(select("进入副本 "+.@md_name$+":取消") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ff与兽人记忆副本相关的所有记录和后续影响均已删除。现在可以重新创建或进入副本。^000000";
		erasequest 12059;
		close;
	}
	mes "创建副本必须是队长，且队伍至少有 2 名成员。";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "发生未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "纪念副本兽人记忆不存在。";
		mes "纪念副本已被队长摧毁或因时限结束。请 2 小时后再试。";
		close;
	case IE_NOMEMBER:
		mes "只有队伍成员才能进入纪念副本。";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +" 队伍成员 "+strcharinfo(0)+" 已进入兽人记忆。",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}
