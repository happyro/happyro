function	script	F_Malaya_Jeepney	{
	.@mapName$        = getarg(0);
	.@passengers      = getarg(1);

	// set the other messages of varying amount)
	for (.@i = 5; .@i < getargcount(); .@i++) {
		setd(".@msgJeepneyInfo$[" + (.@i - 5) + "]", getarg(.@i));
	}

	if (malaya_hi < 10) {
		mes "[吉普尼司机]";
		mes getarg(2);
		close;
	} else if ((malaya_hi >= 10) && (malaya_hi < 20)) {
		mes "[吉普尼司机]";
		mes getarg(3);
		close;
	} else {
		mes "[吉普尼司机]";
		mes getarg(4);
		next;
	}
	switch(select("上车 [乘客 " + getmapusers(.@mapName$) + "/" + .@passengers +"]：吉普尼？:还有其他吉普尼吗？:啊……是的……")) {
	case 1:
		if (getmapusers(.@mapName$) >= .@passengers) {
			mes "[吉普尼司机]";
			mes "恐怕吉普尼已经满员了。";
			mes "抱歉，换乘其他吉普尼怎么样？";
			close;
		} else {
			mes "[吉普尼司机]";
			mes "祝你今天愉快。";
			close2;
			warp .@mapName$,29,24;
			end;
		}
	case 2:
		// iterate through all the jeepney information for this specific NPC
		for (.@i = 0; .@i < getarraysize(.@msgJeepneyInfo$); .@i++) {
			mes "[吉普尼司机]";
			mes getd(".@msgJeepneyInfo$[" + .@i + "]");

			if (.@i < (getarraysize(.@msgJeepneyInfo$) - 1)) {
				next;
			}
		}
		close;
	case 3:
		mes "[吉普尼司机]";
		mes "哦！当然还有其他吉普尼在运行。";
		next;
		mes "[吉普尼司机]";
		mes "马来港有 12、30 和 60 人座的吉普尼，每种各运行 3 辆。";
		next;
		mes "[吉普尼司机]";
		mes "想知道其他车辆的位置吗？";
		next;
		switch (select("我想知道 12 人座车辆的位置。:我想知道 30 人座车辆的位置。:我想知道 60 人座车辆的位置。")) {
		case 1:
			viewpoint 1, 237, 240, 1, 0xF7E009;
			viewpoint 1, 67 , 44 , 2, 0xF7E009;
			viewpoint 1, 282, 129, 3, 0xF7E009;
			set .@zif_in, 12;
			break;
		case 2:
			viewpoint 1, 134, 250, 4, 0xF7E009;
			viewpoint 1, 341, 153, 5, 0xF7E009;
			viewpoint 1, 293, 290, 6, 0xF7E009;
			set .@zif_in, 30;
			break;
		case 3:
			viewpoint 1, 242, 221, 7, 0xF7E009;
			viewpoint 1, 62 , 245, 8, 0xF7E009;
			viewpoint 1, 257, 58 , 9, 0xF7E009;
			set .@zif_in, 60;
			break;
		}
		mes "[吉普尼司机]";
		mes .@zif_in+" 人座吉普尼的位置已标记在你的地图上。";
		next;
		break;
	}
	mes "[吉普尼司机]";
	mes "旅途愉快。";
	close;
}
