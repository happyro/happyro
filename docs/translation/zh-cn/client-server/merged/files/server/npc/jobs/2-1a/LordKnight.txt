valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Lord Knight]";
			mes "恭喜你。";
			mes "向勇士致敬！";
			close;
		}
		mes "[Lord Knight]";
		mes "我们骑士肩负着";
		mes "重大的责任……";
		mes "那就是服务与守护。";
		next;
		mes "[Lord Knight]";
		mes "即使要付出";
		mes "自己的生命，";
		mes "我们也必须守护";
		mes "同伴的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Lord Knight]";
		mes "你的时刻到了！";
		mes "这个世界仍然需要你。";
		mes "请以全新的姿态";
		mes "继续英雄的人生。";
		next;
		mes "[Lord Knight]";
		mes "你想成为";
		mes "骑士领主吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[Lord Knight]";
			mes "准备好后，";
			mes "随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[Lord Knight]";
			mes "你仍然可以学习更多技能。请在回来找我之前，使用";
			mes "完所有剩余的技能点。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Lord Knight]";
		mes "恭喜你！";
		mes "作为骑士领主，";
		mes "我希望你能";
		mes "站在战斗最前线，";
		mes "带领盟友走向胜利！";
		close;
	}
	mes "[Lord Knight]";
	mes "欢迎";
	mes "来到瓦尔哈拉，";
	mes "荣誉大厅。";
	next;
	mes "[Lord Knight]";
	mes "请在这里";
	mes "放松休息。";
	mes "在这里好好休息吧。";
	mes "向勇士致敬！";
	close;
}
