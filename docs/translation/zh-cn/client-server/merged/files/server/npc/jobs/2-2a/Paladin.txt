valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[圣骑士]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[圣骑士]";
		mes "别忘了";
		mes "圣战";
		mes "即将到来！";
		mes "我们必须做好准备！";
		mes "愿正义之光";
		mes "永远照亮道路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[圣骑士]";
		mes "圣战将在";
		mes "我们意识到之前";
		mes "降临。比以往更需要";
		mes "强壮的男女";
		mes "为善与正义";
		mes "而战。";
		next;
		mes "[圣骑士]";
		mes "你愿意作为圣骑士";
		mes "站在正义一方";
		mes "战斗吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[圣骑士]";
			mes "准备好后，";
			mes "随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[圣骑士]";
			mes "你仍然可以学习更多技能。请在回来前使用";
			mes "所有剩余技能点。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[圣骑士]";
		mes "恭喜。";
		mes "作为圣骑士，我希望";
		mes "你保护比你弱小的人，";
		mes "并为我们带来";
		mes "即将到来的";
		mes "善恶之战的胜利。";
		close;
	}
	mes "[圣骑士]";
	mes "欢迎";
	mes "来到英灵殿，";
	mes "荣誉之厅。";
	next;
	mes "[圣骑士]";
	mes "请让自己";
	mes "舒适地";
	mes "待在这里。";
	mes "向勇士致敬！";
	close;
}
