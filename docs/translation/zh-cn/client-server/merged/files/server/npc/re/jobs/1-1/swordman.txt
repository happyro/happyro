izlude_in,74,172,4	script	Swordman#swd	119,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin)) {
			mes "[Swordman]";
			mes "这……";
			mes "不可能……";
			mes "您已经转生了，对吗？";
			next;
			mes "[Swordman]";
			mes "看来您又踏上了剑士之路！一旦习惯挥剑，就再也回不去了！！";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Swordman]";
				mes "嗯？啊，成为剑士前必须先掌握基础技能。";
				next;
				mes "[Swordman]";
				mes "完成新手基础技能学习后再来找我。";
				close;
			}
			mes "[Swordman]";
			mes "太好了！我马上为您转职为剑士！";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Swordman]";
			mes "嗯……您看起来是一位经验丰富的剑士。不过，仍然要继续训练提升技能和力量！";
			close;
		}else{
			mes "[Swordman]";
			mes "嗯……？";
			mes "您是一位";
			mes "转生的战士，对吗？";
			next;
			mes "[Swordman]";
			mes "嗯……";
			mes "看来成为剑士";
			mes "并不是您的命运。抱歉，";
			mes "我似乎什么也";
			mes "帮不了您。";
			mes "";
			close;
		}
	}
	mes "[Swordman Guildsman]";
	mes "这里是剑士公会。";
	mes "您为什么来这里？";
	next;
	switch(select("告诉我剑士的事。:我想成为剑士。:没什么。")) {
	case 1:
		mes "[Swordman Guildsman]";
		mes "您想进一步了解强大的剑士？好的！";
		next;
		mes "[Swordman Guildsman]";
		mes "剑士最显著的特点是在近战中发挥真正的实力。";
		mes "原因有三个！";
		next;
		mes "[Swordman Guildsman]";
		mes "第一，剑士比其他职业拥有更高的 HP。";
		mes "第二，除了弓和杖，剑士可以使用其他所有武器，发挥最佳战斗力。";
		mes "第三，大多数剑士技能都能发动强力的物理攻击。";
		next;
		mes "[Swordman Guildsman]";
		mes "虽然解释得很简单，但我相信您已经理解剑士的核心意义。";
		mes "在我看来，剑士是最棒的职业！";
		close;
	case 2:
		mes "[Swordman Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Swordman) {
				mes "您已经是一名优秀的剑士了，对吗？";
				mes "继续努力成为伟大的剑士吧。";
			}else{
				mes "您已经拥有其他职业了，对吗？";
				mes "这个玩笑开得太过分了。";
			}
			close;
		}else{
			if (!callfunc("F_CanChangeJob")) {
				mes "很抱歉，想成为剑士，必须达到至少 ^4A4AFF职业等级 10^000000";
				mes "以及 ^4A4AFF基础技能等级 9^000000。";
				next;
				mes "[Swordman Guildsman]";
				mes "未满足最低要求也想成为剑士？";
				mes "您以为成为剑士那么容易吗？";
				close;
			}
			mes "嗯，您的职业等级和基础技能等级都符合要求。";
			mes "很好。现在要成为剑士吗？";
			next;
			if(select("是的。:我再考虑一下。") == 2) {
				mes "[Swordman Guildsman]";
				mes "好的。选择职业需要谨慎决定。";
				mes "不过真遗憾……您克服了所有困难后却又要重新考虑……";
				close;
			}
			mes "[Swordman Guildsman]";
			mes "恭喜！从现在起，您将开启剑士的人生！";
			mes "现在就开始吧！";
			next;
			mes "[Swordman Guildsman]";
			mes "再次恭喜您成为剑士，希望您积极参加活动，为公会复兴贡献力量。";
			callfunc "Job_Change",Job_Swordman;
			getitem 13415,1; // N_Falchion
			close;
		}
	case 3:
		close;
	}
}
