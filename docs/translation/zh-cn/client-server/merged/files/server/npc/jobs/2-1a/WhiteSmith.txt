	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神工匠]";
			mes "恭喜。";
			mes "向战士致敬！";
			close;
		}
		mes "[神工匠]";
		mes "啊……";
		mes "工艺的巅峰。";
		mes "";
		mes "那就是";
		mes "神工匠的作品。";
		next;
		mes "[神工匠]";
		mes "当你成为";
		mes "最优秀中的最优秀者，";
		mes "你会开始锻造，目标是";
		mes "发现";
		mes "更新更好的";
		mes "制作方法……";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[神工匠]";
		mes "时刻到了！";
		mes "我们的世界需要勇敢、";
		mes "勤奋的冒险者，";
		mes "像你一样……";
		next;
		mes "[神工匠]";
		mes "你愿意成为";
		mes "神工匠吗？";
		next;
		if (select("不:是") == 1) {
			mes "[神工匠]";
			mes "准备好后";
			mes "随时回来。";
			mes "向战士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[神工匠]";
			mes "你还可以学习更多技能。请在回来前用完";
			mes "所有剩余技能点。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[神工匠]";
		mes "恭喜！";
		mes "作为神工匠，";
		mes "希望你能为米德加尔特";
		mes "锻造通往更光明";
		mes "未来的道路。";
		close;
	}
	mes "[神工匠]";
	mes "欢迎";
	mes "来到英灵殿，";
	mes "荣誉大厅。";
	next;
	mes "[神工匠]";
	mes "请在这里";
	mes "安心休息。";
	mes "趁你在这里。";
	mes "向战士致敬！";

}
