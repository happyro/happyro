payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "嘿，我认识你。";
			mes "你以前参加过这个测试，";
			mes "对吧？";
			next;
			mes "[Archer Guildsman]";
			mes "啊，你一定去过英灵殿并转生了。";
			mes "哇，真令人佩服！";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "呃……";
				mes "成为弓箭手前，最好先学会所有基础技能。";
				next;
				mes "[Archer Guildsman]";
				mes "好，回头见。";
				close;
			}
			mes "[Archer Guildsman]";
			mes "那么，我就不多说了。";
			mes "我知道你会成为优秀的弓箭手……";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "这次没有特别奖励，希望你能理解。请保重。";
			mes "";
			close;
		}
		else {
			mes "[Archer Guildsman]";
			mes "哦……？";
			mes "嘿，你在这里做什么……？";
			next;
			mes "[Archer Guildsman]";
			mes "我看得出你不适合当弓箭手，感觉你应该做别的事……";
			mes "";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "你好。有什么我能帮你的吗？";
	next;
	switch(select("我想成为弓箭手。:请告诉我要求。:没什么，谢谢。")) {
	case 1:
		if (BaseJob == Job_Archer) {
			mes "[Archer Guildsman]";
			mes "你已经成为弓箭手了……";
			close;
		}
		else if (BaseJob != Job_Archer && BaseJob != Job_Novice) {
			mes "[Archer Guildsman]";
			mes "嗯……";
			mes "你看起来一点也不像初心者……";
			next;
			mes "[Archer Guildsman]";
			mes "不管怎样，你已经有职业，不能再选择弓箭手。";
			close;
		}
		if (job_archer_q == 0) {
			mes "[Archer Guildsman]";
			mes "你想成为弓箭手吗？";
			mes "如果想，就填写这份申请表。";
			mes "";
			next;
			if (select("申请。:取消") == 1) {
				set job_archer_q,1;
				setquest 1004;
				mes "[Archer Guildsman]";
				mes "好，在这里签名。满足要求后我就会让你转职。";
				next;
				mes "[Archer Guildsman]";
				mes "如果你觉得已经满足要求，我们现在可以检查。准备好了吗？";
				mes "";
				next;
				if (select("是的。:还没有。") == 1) {
					mes "[Archer Guildsman]";
					mes "好，让我检查一下。";
					next;
				}
				else {
					mes "[Archer Guildsman]";
					mes "明白了。如果想查看要求，请自便。";
					close;
				}
			}
			else {
				mes "[Archer Guildsman]";
				mes "好吧。";
				mes "下次见。";
				close;
			}
		}
		mes "[Archer Guildsman]";
		mes "你是……" + strcharinfo(0) + "吗？";
		next;
		if (!callfunc("F_CanChangeJob")) {
			mes "[Archer Guildsman]";
			mes "你的职业等级还不够。请再次确认要求。";
			next;
			mes "[Archer Guildsman]";
			mes "职业等级必须至少为10，而且别忘了学会所有基础技能。完成后再回来。";
			close;
		}
		if (job_archer_q == 1) {
			set .@archer_item1,countitem(1066) * 5;
			set .@archer_item2,countitem(1067) * 3;
			set .@archer_item3,countitem(1068) * 2;
			set .@archer_item4,countitem(1019);
			set .@total_archer,.@archer_item1 + .@archer_item2 + .@archer_item3 + .@archer_item4;
			set .@total_archer2,(((.@archer_item2 + .@archer_item3) * 2) + .@archer_item4);
			mes "[Archer Guildsman]";
			mes "太棒了！";
			mes "那么，";
			mes "let's see...";
			next;
			mes "[Archer Guildsman]";
			mes "我会评估你带来的各种木材价值，这些木材将用于制作弓。";
			next;
			if (countitem(1066) == 0 && countitem(1067) == 0 && countitem(1068) == 0 && countitem(1019) == 0) {
				mes "[Archer Guildsman]";
				mes "Um...";
				mes "很遗憾，你没有带来所需物品。我无法进行评估。";
				close;
			}
			mes "[Archer Guildsman]";
			if (countitem(1066) != 0) mes " S级：" + countitem(1066) + "个，评分：" + .@archer_item1 + "。 ";
			if (countitem(1067) != 0) mes " A级：" + countitem(1067) + "个，评分：" + .@archer_item2 + "。 ";
			if (countitem(1068) != 0) mes " B级：" + countitem(1068) + "个，评分：" + .@archer_item3 + "。 ";
			if (countitem(1019) != 0) mes " C级：" + countitem(1019) + "个，评分：" + .@archer_item4 + "。 ";
			if (.@total_archer < 25) {
				mes "总评分：^FF0000" + .@total_archer + "^000000 / 40";
			mes "";
				next;
				mes "[Archer Guildsman]";
				mes "低于25！？评分必须至少达到25！加油，再努力些！";
				close;
			}
			else {
				mes "总评分：^0000FF" + .@total_archer + "^000000 / 40";
			mes "";
				next;
				mes "[Archer Guildsman]";
				if (.@total_archer > 40) {
					mes "哇！超过40分！";
					mes "太棒了！恭喜！";
				}
				else if (.@total_archer > 30) {
					mes "超过30分！干得漂亮！";
					mes "恭喜！";
				}
				else {
					mes "*叹气* 你勉强通过了……总之，做得不错。";
				}
			}
			next;
			mes "[Archer Guildsman]";
			mes "我会把这些木材交给弓制作部门。既然你满足要求，我现在就让你转职！";
			if (countitem(1066) != 0) delitem 1066,countitem(1066); //Tree_Of_Archer_1
			if (countitem(1067) != 0) delitem 1067,countitem(1067); //Tree_Of_Archer_2
			if (countitem(1068) != 0) delitem 1068,countitem(1068); //Tree_Of_Archer_3
			if (countitem(1019) != 0) delitem 1019,countitem(1019); //Wooden_Block
		}
		next;
		callfunc "Job_Change",Job_Archer;
		callfunc "F_ClearJobVar";
		completequest 1004;
		mes "[Archer Guildsman]";
		mes "恭喜！";
		mes "你现在是弓箭手了！";
		next;
		mes "[Archer Guildsman]";
		mes "当然，希望你用自己的努力为弓箭手公会的未来作出贡献。";
		next;
		mes "[Archer Guildsman]";
		mes "啊，你的弓已经从弓制作部门送来了。给你，收下吧！它属于你～";
		getitem 1702,1; //Bow_
		getitem 1750,.@total_archer2; //Arrow
		next;
		mes "[Archer Guildsman]";
		mes "现在出发吧。带着自豪去狩猎，你可是由最优秀的人训练的！";
		close;
	case 2:
		mes "[Archer Guildsman]";
		mes "我来说明成为弓箭手的要求。";
		if (BaseJob != Job_Novice) {
			if (BaseJob == Job_Archer) {
				next;
				mes "[Archer Guildsman]";
				mes "But...";
				mes "你已经是弓箭手了，应该已经知道这些……";
			}
			else {
				next;
				mes "[Archer Guildsman]";
				mes "等一下。你已经选择了其他职业，不需要知道这些～";
			}
			mes "所以……嗯……没必要告诉你要求了……";
		}
		next;
		mes "[Archer Guildsman]";
		mes "首先，初心者职业等级必须达到9，并学会所有基础技能。";
		next;
		mes "[Archer Guildsman]";
		mes "弓箭手需要极高的集中力和反应力，所以我们不接收缺乏耐心的人。";
		next;
		mes "[Archer Guildsman]";
		mes "你还必须收集^FF0000木材^000000。木材有4种，品质各不相同。我们会根据品质为木材评分。";
		next;
		mes "[Archer Guildsman]";
		mes "要成为弓箭手，你必须在40分中获得至少^0000FF25^000000分。可以从‘柳树’上获得木材。不过要小心，它们可能是强大的怪物。";
		mes "";
		close;
	case 3:
		close;
	}
}
