-	script	::Airship_Staff_izlude	-1,{
	mes "[飞艇工作人员]";
	mes "欢迎！您想要搭乘";
	mes "前往朱诺和拉赫尔的";
	mes "飞艇吗？";
	next;
	if (select("搭乘前往朱诺/拉赫尔的飞艇。:取消。") == 1) {
		mes "[飞艇工作人员]";
		mes "登船费用为 1,200 Zeny。";
		mes "不过，使用飞艇免费券即可免除费用。";
		mes "您还要登船吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[飞艇工作人员]";
			mes "抱歉，您的 Zeny 不足";
			mes "以支付 1,200 Zeny 的登船费。";
			close;
		}
	}
	mes "[飞艇工作人员]";
	mes "感谢您的光临，欢迎再次前来。";
	mes "祝您今天愉快～";
	close;
}
