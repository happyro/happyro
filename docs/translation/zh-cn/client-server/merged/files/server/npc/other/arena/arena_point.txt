//= Point exchanger: Turbo track points -> Arena Points

prt_are_in,103,11,3	script	Arena Point Manager	732,{
	mes "[竞技场点数管理员]";
	mes "希望你在竞技场玩得";
	mes "开心。如果你在";
	mes "艾尔迪巴兰加速赛道获得了";
	mes "加速赛道点数，我可以";
	mes "将其兑换为竞技场点数。";
	next;
	switch(select("点数查询:兑换点数:^660000兑换说明^000000")) {
	case 1:
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + ",";
		mes "你目前拥有";
		mes "" + arena_point + " 点竞技场点数";
		mes "以及 " + tt_point + " 点加速赛道点数。";
		close;
	case 2:
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + ",";
		mes "你目前拥有";
		mes "" + arena_point + " 点竞技场点数";
		mes "以及 " + tt_point + " 点加速赛道点数。";
		next;
		mes "[竞技场点数管理员]";
		mes "请选择";
		mes "加速赛道点数到竞技场点数的";
		mes "兑换方式。请注意，";
		mes "兑换超过 10 点时只能以";
		mes "^4D4DFF10 的倍数^000000 兑换。";
		next;
		switch(select("2 TP -> 1 AP:4 TP -> 2 AP:6 TP -> 3 AP:8 TP -> 4 AP:10 TP and more:Cancel")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[竞技场点数管理员]";
			mes "请输入";
			mes "兑换次数。";
			mes "将 10 点加速赛道点数兑换为";
			mes "竞技场点数，最大";
			mes "可输入 20。";
			mes "取消请输入“^3355FF0^000000”。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[竞技场点数管理员]";
				mes "你有";
				mes "取消了";
				mes "此服务。";
				close;
			}
			else if (.@input > 20) {
				mes "[竞技场点数管理员]";
				mes "请求超过";
				mes "最大限制。请";
				mes "输入不超过 20 的数值。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[竞技场点数管理员]";
					mes "你将超过";
					mes "竞技场点数";
					mes "上限。继续兑换";
					mes "这些";
					mes "加速赛道点数将使竞技场点数超过";
					mes "29,000 点上限。";
					next;
					mes "[竞技场点数管理员]";
					mes "请先消费部分";
					mes "竞技场点数后再";
					mes "使用此服务。";
					mes "感谢惠顾。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[竞技场点数管理员]";
					mes "抱歉，";
					mes "你的加速赛道点数不足，无法";
					mes "进行";
					mes "此次竞技场";
					mes "点数兑换。";
					close;
				}
				else {
					mes "[竞技场点数管理员]";
					mes "你已完成兑换";
					mes "10 点加速赛道点数";
					mes "为 5 点竞技场点数，次数：" + .@input + " 次。";
					mes "总计 " + .@want_point1 + " 点加速赛道点数兑换为";
					mes "" + .@want_point +" 点竞技场点数。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[竞技场点数管理员]";
					mes "" + strcharinfo(0) + ",";
					mes "你现在拥有";
					mes "^4682B4" + arena_point + "^000000 点竞技场点数";
					mes "以及 ^00688B" + tt_point + "^000000 点加速赛道点数。";
					mes "感谢惠顾。";
					close;
				}
			}
		case 6:
			mes "[竞技场点数管理员]";
			mes "你有";
			mes "取消了";
			mes "此服务。";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[竞技场点数管理员]";
		mes "你将超过";
		mes "竞技场点数";
		mes "上限。继续兑换";
		mes "这些";
		mes "加速赛道点数将使竞技场点数超过";
		mes "29,000 点上限。";
		next;
		mes "[竞技场点数管理员]";
		mes "请先消费部分";
		mes "竞技场点数后再";
		mes "使用此服务。";
		mes "感谢惠顾。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + ",";
		mes "你已将 "+ .@track_point +" 点加速赛道点数兑换为 "+ .@arena_point +" 点竞技场点数。";
		mes "兑换完成。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + ",";
		mes "你现在拥有";
		mes "^4682B4" + arena_point + "^000000 点竞技场点数";
		mes "以及 ^00688B" + tt_point + "^000000 点加速赛道点数。";
		mes "感谢惠顾。";
		close;
	}
	else {
		mes "[竞技场点数管理员]";
		mes "抱歉，你没有足够的";
		mes "加速赛道点数。";
		mes "使用此兑换服务至少需要 2 点加速赛道点数。";
		close;
	}
}
