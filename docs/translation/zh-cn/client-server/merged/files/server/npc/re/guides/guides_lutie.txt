xmas,140,137,3	script	Lutie Guide#xmas	717,{
	mes "[Lutie Guide]";
	mes "欢迎来到^8B4513鲁提耶^000000，";
	mes "这座雪之村。";
	mes "需要我帮你在村里导航吗？";
	F_Navi("[Lutie Guide]");
	while (1) {
		next;
		switch(select("[ 礼品店 ]:[ 教堂 ]:[ 武器与防具店 ]:[ 道具店 ]:[ 圣诞老人的家 ]:[ 玩具工厂 ]:移除小地图标记:取消")) {
		case 1:
			callsub L_Mark, F_Navi("礼品店","xmas,172,131","^008080");
			viewpoint 1,172,131,2,0x008080;
			break;
		case 2:
			callsub L_Mark, F_Navi("教堂","xmas,104,287","^006400");
			viewpoint 1,104,287,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("武器与防具店","xmas,171,158","^FF1493");
			viewpoint 1,171,158,3,0xFF1493;
			break;
		case 4:
			callsub L_Mark, F_Navi("道具店","xmas,122,131","^8B4513");
			viewpoint 1,122,131,4,0x8B4513;
			break;
		case 5:
			callsub L_Mark, F_Navi("圣诞老人的家","xmas,149,237","^B9062F"),1;
			viewpoint 1,149,237,5,0x9400D3;
			break;
		case 6:
			callsub L_Mark, F_Navi("玩具工厂","xmas,143,312","^9400D3");
			viewpoint 1,143,312,6,0xFF0000;
			break;
		case 7:
			mes "[Lutie Guide]";
			mes "好的，我会移除小地图上的所有标记。";
			mes "还有什么我能帮你的吗？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			break;
		case 8:
			mes "[Lutie Guide]";
			mes "旅途中请注意安全。";
			close;
		}
	}
	end;
L_Mark:
	mes "[Lutie Guide]";
	mes "让我在小地图上标记"+((getarg(1,0))?"":"该地点");
	mes getarg(0)+"^000000";
	mes "的位置。";
	mes "还想查看其他地点吗？";
	return;
}
