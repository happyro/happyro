malaya,71,72,4	script	Port Malaya Guide#01::GuideMalaya	579,{
	mes "[马来港向导]";
	mes "欢迎来到^8B4513马来港^000000，这座会呼吸的村庄。";
	mes "这里的文化与传说都栩栩如生！";
	mes "需要我帮你在村子里导航吗？";
	F_Navi("[马来港向导]");
	while(1) {
		next;
		switch(select("[ 卡普拉职员 ]:[ 旅馆与酒吧 ]:[ 武器与防具商店 ]:[ 工具商店 ]:[ 医院 ]:[ 公园 ]:[ 港口 ]:移除小地图标记:取消")) {
		case 1:
			callsub L_Mark, "^0000FF卡普拉职员^000000",1;
			viewpoint 1,71,79,0,0x0A82FF;
			viewpoint 1,234,204,1,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("旅馆与酒吧","malaya,193,215","^006400");
			viewpoint 1,193,215,2,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("武器与防具商店","malaya,114,212","^008080");
			viewpoint 1,114,212,3,0x4F543C;
			break;
		case 4:
			callsub L_Mark, F_Navi("工具商店","malaya,297,167","^FF1493");
			viewpoint 1,297,167,4,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("医院","malaya,50,75","^8B4513");
			viewpoint 1,50,75,5,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("公园","malaya,280,371","^8B4513");
			viewpoint 1,280,371,6,0x8B4513;
			break;
		case 7:
			callsub L_Mark, F_Navi("港口","malaya,276,55","^8A2BE2");
			viewpoint 1,276,55,7,0x8A2BE2;
			break;
		case 8:
			mes "[马来港向导]";
			mes "我会移除你小地图上的所有标记。";
			mes "还有什么我能帮你的吗？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 9:
			mes "[马来港向导]";
			mes "祝你旅途愉快。";
			close;
		}
	}
	end;
L_Mark:
	mes "[马来港向导]";
	mes "你的"+getarg(0);
	mes "已在小地图上标记。";
	mes "请查看小地图。";
	mes "还有什么我能帮你的吗？";
	return;
}
