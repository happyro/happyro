		setquest 16578;

		next;

	else if (ep18_main == 46) {
		cutin "ep18_maram_01.png",2;
		mes "[玛拉姆]";
		mes "尽量避开警戒塔进去吧。";
		next;

	next;
	// player has party
	if (is_party_leader())
		// player is leader of the party
		.@create$ = "申请进入" + .@md_name$;

	switch( select( .@create$, "进入" + .@md_name$ ) ) {
	case 1:
		cutin "ep18_maram_01.png",2;
		mes "[玛拉姆]";
		mes "申请完成后，再和我交谈并选择^0000CD" + .@md_name$ + "入口^000000。";
		if (instance_create(.@md_name$) < 0) {
			mes "队伍：" + getpartyname(getcharid(1)) + "";
			mes "队长：" + strcharinfo(0) + "";
			mes "^0000ff" + .@md_name$ + " ^000000- 未知错误";
			close3;
		}
		close3;
	case 2:
		// Note: instance not forced solo
		switch( instance_enter(.@md_name$) ) {
		case IE_OTHER:
			mes "[玛拉姆]";
			mes "^ff0000发生未知错误。^000000";
			close;
		case IE_NOINSTANCE:
			cutin "ep18_maram_01.png",2;
			mes "[玛拉姆]";
			mes "你似乎还没有创建副本。请稍后再确认。";
			close3;
		case IE_NOMEMBER:
			mes "[玛拉姆]";
			mes "^ff0000只有队伍成员才能进入纪念副本。^000000";
			close;
		case IE_OK:
			mapannounce "que_thr", "队伍的" + strcharinfo(0) + "正在进入" + .@md_name$ + "。", bc_map, 0xFF99;
			// warp 1@tcamp,103,237;
			end;
		}
	}
	end;

OnInit:
	questinfo( QTYPE_QUEST2, QMARK_YELLOW, "isbegin_quest(16577) == 1" );
	end;
