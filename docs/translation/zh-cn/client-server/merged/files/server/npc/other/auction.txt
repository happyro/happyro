function	script	F_AuctionWarper	{
	mes "[Auction Hall Guide]";
	mes "你好，要进入拍卖大厅吗？";
	next;
	if (select("Yes:No") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "[Auction Hall Guide]";
			mes "太好了！希望你玩得开心，享受拍卖～";
		} else {
			mes "[Auction Hall Guide]";
			mes "祝你拍卖愉快。";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "[Auction Hall Guide]";
	mes "好吧，回头见。改变主意的话，欢迎来享受拍卖～";
	close;
}
