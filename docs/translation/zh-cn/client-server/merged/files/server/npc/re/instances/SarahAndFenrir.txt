	case -1:

	case 1:
		mes "- 次元装置已关闭。";
		mes "看来无法进入。-";
		close;
	case 2:
		break;
	}
	if (isbegin_quest(9335) > 0) {
		mes "- 次元装置已关闭。";
		mes "看来无法进入。-";
		close;
	}

			erasequest 9336;
		// warp "1@glast",367,304;
