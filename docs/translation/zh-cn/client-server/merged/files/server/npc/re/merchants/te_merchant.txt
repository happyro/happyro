	next;
	for ( 。@i = 1; 。@i < getargcount(); 。@i += 2 ) {
		if (rentalcountitem(getarg(。@i)) > isequippedcnt(getarg(。@i)))// don't count item equipped
			。@menu$ = 。@menu$ + "^4d4dff"+ getarg(。@i+1) +" - 已租用^000000:";
		else
			。@menu$ = 。@menu$ + getarg(。@i+1) + ":";
	}
	。@total_item = (getargcount() -1) /2;
	。@s = select( 。@menu$ + ":View other items" ) -1;
	if (。@s <= 。@total_item) {
		。@index = 。@s *2 +1;
		mes "[租赁物品管理员]";
		if (rentalcountitem(getarg(。@index)) > isequippedcnt(getarg(。@index)))
			mes "你已经有"+ getarg(。@index+1) +"。";
		else {
			mes "这是"+ getarg(。@index+1) +"。";
			mes "^4d4dffThis item is effective for the next 24 hours^000000 and after that it expires。";
			rentitem getarg(。@index),86400;
		}
		mes "还需要其他东西吗？";
		next;
	}
	return;
