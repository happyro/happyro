//===== rAthena Script =======================================
//= Insurance Exchanger
//===== Description: =========================================
//= Exchange items for New_Insurance item.
//===== Changelogs: ==========================================
//= 1.0 Initial release. [Balfear]
//============================================================

prontera,82,108,5	script	Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "你携带的物品数量过多，无法继续对话。";
		mes "请整理物品后再试。";
		close;
	}
	mes "[心之商人]";
	mes "你好！看得出来你是把心交给旅行、乘船横渡大陆、享受危险刺激的冒险者！";
	next;
	if(!countitem(.@old_item)){
		mes "[心之商人]";
		mes "你储物箱里有" + mesitemlink( .@old_item ) + "吗？";
		next;
		mes "[心之商人]";
		mes "如果你带来一个" + .@old_item_name$ + "，我会把它兑换成" + mesitemlink( .@new_item ) + ". What do you think? 你储物箱里有idea what this item is?";
	} else {
		mes "[心之商人]";
		mes "你有没有想过把你的" + mesitemlink( .@old_item ) + "换成更好的东西？比如……" + mesitemlink( .@new_item ) + ". What do you think?";
	}
	next;
	switch(select("这是什么？",(countitem(.@old_item)?"请帮我兑换！":""),"我没兴趣")) {
	case 1:
		mes "[心之商人]";
		mes mesitemlink( .@old_item ) + "是一件使用后 30 分钟内可防止死亡时损失经验值一次的物品。";
		next;
		mes "[心之商人]";
		mes "不过很多人都对时间限制感到遗憾，对吧？所以我研究了一下，准备了" + mesitemlink( .@new_item ) + ".";
		next;
		mes "[心之商人]";
		mes mesitemlink( .@new_item ) + "是一件只要持有就能防止死亡时损失经验值一次的物品，^EE0000无论持续时间^000000. 当然，死亡后，"+.@new_item_name$+"会被消耗。";
		next;
		select("为什么要兑换？");
		mes "[心之商人]";
		mes "你对此好奇吗？我是商人。无论关乎生还是死，买卖就是我的宿命。你觉得我会错过赚钱的机会吗？我赚到钱，活着的人也能减轻死亡的痛苦。这不是很划算吗？";
		next;
		mes "[心之商人]";
		mes "兑换比例是5 " + mesitemlink( .@old_item ) + "换 1 个" + mesitemlink( .@new_item ) + ". 你储物箱里有questions?";
		next;
		mes "[心之商人]";
		mes "嗯~你觉得呢？这不是公平的交易吗？你想一直担心时间吗？这是享受惊险冒险的冒险者必备之物。";
		close;
		
	case 2:	
		mes "[心之商人]";
		mes "对，对。明智的决定。换成更好的东西不是更好吗？";
		next;
		mes "[心之商人]";
		mes "我可以把你的^0000cd5 "+.@old_item_name$+"兑换成 1 个"+.@new_item_name$+".";
		next;
		mes "[心之商人]";
		mes "不过，^EE0000[Not for sale] " + .@old_item_name$ + "不符合兑换条件^000000, 所以带来这里要求兑换也是没用的。";
		next;
		mes "[心之商人]";
		mes "现在兑换吗？";
		next;
		if (select("请帮我兑换！:我不想兑换。") == 2)
			break;
		mes "[心之商人]";
		mes "你想兑换多少？^0000cd一次最多可兑换 10 个"+.@new_item_name$+"^000000.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[心之商人]";
			mes "嗯？你没兴趣吗？这是很棒的商品，非常适合你这样的人。不过如果你不想保护生命，我也不会强迫你。";
			close;
		}
		if (.@amount > 10) {
			mes "[心之商人]";
			mes "正如我说的，^0000cd一次最多只能兑换 10 个^000000. Check the quantity again.";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[心之商人]";
			mes "呃，抱歉……你的" +.@old_item_name$+ "不够，对吧？兑换比例是 5 个" + mesitemlink( .@old_item ) + "换 1 个" + mesitemlink( .@new_item ) + "。请检查数量。";
			close;
		}
		mes "[心之商人]";
		mes "现在，我已将你的^0000cd" + .@old_item_name$ + "^000000兑换为^EE0000" + .@new_item_name$ + "^000000！这笔交易很划算，对吧？下次你拿到"+.@old_item_name$+"，再来找我！";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[心之商人]";
	mes "这是很划算的交易！冒险者，这是在危险之地冒险的人必备的物品！";
	next;
	mes "[心之商人]";
	mes "如果你感兴趣，记得回来兑换。好吗？";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "你有好东西。想把你的" + getitemname(12209) + "换成更好的吗？", "", bc_self;
	end;
}
