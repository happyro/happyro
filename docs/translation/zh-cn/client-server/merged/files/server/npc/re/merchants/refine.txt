//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "我是最优秀的武器匠！";
	mes "我不强化普通又无聊的物品。";
	mes "我只强化 10 级以上的物品。";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "总之，如果你的物品达到 10 级以上，就可以使用我的服务。";
	mes "你想让我强化什么？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想你没有可以强化的物品……";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "那里没有装备";
		mes "我可以";
		mes "强化的物品。";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我完全不能";
		mes "强化这件物品……";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我说过，我不处理低于 10 级的物品。";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "我不能再强化这件物品了。";
		mes "它已经达到";
		mes "强化极限！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "嗯，" + .@article$ + " " + .@type$ + "，可以吗？";
	mes "如果你想强化这件" + .@type$ + "，";
	mes "我需要 1 个^003366" + getitemname(.@material) + "^000000和" + callfunc("F_InsertComma",.@price) + " Zeny。";
	mes "确定要继续吗？";
	next;
	if (select("是:否") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……如果你介意……算了……";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "这件"+.@type$+"已经强化过很多次了。";
			mes "如果再次尝试，可能会损坏。";
			mes "虽然不一定会损坏，但确实存在小概率。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你可能会^FF0000降低这件"+.@type$+"的强化等级^000000，";
			mes "如果它损坏，添加的^FF0000卡片^000000或特殊属性也会全部丢失。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你还要我继续强化吗？";
			mes "我想我已经充分提醒过你了。";
			next;
			if(select("是。:否。") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "好吧，不接受挑战也是一种选择……";
				mes "没有风险……也许这样更明智。";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "嗯。你的钱或"+getitemname(.@material)+"似乎不够。";
			mes "准备好后再回来吧。";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你以为我傻吗？！";
			mes "你趁我没注意换了物品！滚出去！";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
		mes "铿！铿！铿！铿！";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
		mes "很好！成功了！！！";
		mes "我是最优秀的武器匠。";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "铿！铿！铿！铿！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "强化等级下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "铿！铿！铿！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "糟糕！我强化失败了……";
				mes "我不是故意的！";
			}
			mes "即使我是最优秀的武器匠，也可能犯错。";
			mes "这只是命中注定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我会做得更好！别担心！";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以把它强化到安全上限，或者强化指定次数。由你决定。";
		next;
		.@menu2 = select("请强化到安全上限。","我来决定次数。","我改变主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "你想让我强化物品多少次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
		mes "我不能强化这件物品这么多次。";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
		mes "这会尝试将装备强化到超过安全上限" + .@refinecheck + "次。装备可能会损坏……可以吗？";
			next;
			if(select("是……","否……") == 2){
				mes "["+ .@npc_name$ +"]";
			mes "这是你说的……那就这样吧。";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "这是你说的……那就这样吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "这需要" + .@refinecnt + " 个" + getitemname(.@material) + "和" + .@fullprice + " Zeny。可以吗？";
	next;
	if(select("是:否……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "这是你说的……那就这样吧。";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
	mes "嗯。你的钱或"+getitemname(.@material)+"似乎不够。";
		mes "准备好后再回来吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "看这里……你没有装备任何物品……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "铿……不，你以为我会这么蠢吗？！";
			mes "你换掉了它……";
			mes "在我用锤子把你打晕前滚出去！！";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "铿！铿！铿！铿！";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
		mes "铿！铿！铿！铿！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "强化等级下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "铿！铿！铿！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "糟糕！我强化失败了……";
				mes "我不是故意的！";
			}
			mes "即使我是最优秀的武器匠，也可能犯错。";
			mes "这只是命中注定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我会做得更好！别担心！";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "很好！成功了！！！";
	mes "我是最优秀的铁匠。";
	close;
}

	delitem .@material,1;
	Zeny = Zeny-.@price;

	// anti-hack
