		if (.@karma_d > 4) {
			mes "[Scholar]";
			mes "恭喜你。";
			mes "向勇士致敬！";
			close;
		}
		mes "[Scholar]";
		mes "这需要一生……";
		mes "确实需要一生的时间";
		mes "才能积累";
		mes "成为学者所需的";
		mes "知识……";
		next;
		mes "[Scholar]";
		mes "这令人不知所措。";
		mes "你学得越多，";
		mes "就越会发现";
		mes "自己不知道的事情。";
		mes "学习的过程";
		mes "永无止境……";
		close;
	}
	else if (ADVJOB == Job_Professor && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Scholar]";
		mes "米德加尔特的学者";
		mes "还不够多，无法";
		mes "帮助开启繁荣的新时代。";

		mes "这份重大的责任吗？";
		mes "你愿意作为学者";
		mes "为米德加尔特效力吗？";

			mes "[Scholar]";
			mes "准备好后，";
			mes "随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[Scholar]";
			mes "你仍然可以学习更多技能。请在回来找我之前，使用";
			mes "完所有剩余的技能点。";
			close;
		}

		set ADVJOB,0;
		mes "[Scholar]";
		mes "恭喜你！";
		mes "作为教授，我希望";
		mes "你能积极参与";

	mes "荣誉大厅。";

	mes "[Scholar]";
	mes "请在这里";
	mes "放松休息。";
	mes "在这里好好休息吧。";
	mes "向勇士致敬！";

}
