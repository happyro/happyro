valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[吟游诗人]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[吟游诗人]";
		mes "你想";
		mes "和我一起唱歌吗？";
		mes "沙啦啦啦啦~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[吟游诗人]";
		mes "沉闷的凡人世界";
		mes "需要更多";
		mes "欢快的歌曲。";
		mes "你会带给他们";
		mes "扭转";
		mes "对抗邪恶的战局吗？";
		next;
		mes "[吟游诗人]";
		mes "你愿意这样做";
		mes "为了米德加尔特……";
		mes "作为吟游诗人？";
		next;
		if (select("不。:是。") == 1) {
			mes "[吟游诗人]";
			mes "准备好后，";
			mes "随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[吟游诗人]";
			mes "你仍然可以学习更多技能。请在回来前使用";
			mes "所有剩余技能点。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[吟游诗人]";
		mes "恭喜！";
		mes "作为吟游诗人，你的";
		mes "歌曲会给盟友带来";
		mes "希望，给敌人带来";
		mes "绝望。";
		close;
	}
	mes "[吟游诗人]";
	mes "欢迎";
	mes "来到英灵殿，";
	mes "荣誉之厅。";
	next;
	mes "[吟游诗人]";
	mes "请让自己";
	mes "舒适地";
	mes "待在这里。";
	mes "向勇士致敬！";
	close;
}
