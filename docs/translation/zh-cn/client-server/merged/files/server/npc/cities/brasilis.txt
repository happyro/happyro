-	script	::Crewman_bra2	-1,{
	mes "[船员]";
	mes "嘿，你听说过一个叫巴西利斯的地方吗？";
	mes "那是一座像沙漠一样炎热却又多雨的热带城市，非常神秘。";
	next;
	mes "[船员]";
	mes "我们最近发现了一条新的海路，可以轻松到达那里。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "往返只要 10,000 Zeny，VIP 只要 1,000！你要去吗？"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "往返只要 10,000 Zeny！你要去吗？";
	}
	next;
	switch(select("带我去巴西利斯！:我留在这里。")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[船员]";
			mes "我说的是 "+callfunc("F_InsertComma",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[船员]";
			mes "太棒了~！！出发吧~！";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[船员]";
		mes "如果你感兴趣，告诉我一声，我可以带你去。";
		close;
	}
}

		close;
	}
}

brasilis,137,77,5	script	Ice-Cream Maker	85,{
	mes "[冰淇淋商人]";
	mes "来吧~来吧~";
	mes "冰淇淋是炎热天气的完美零食~";
	mes "只要^3355FF100 Zeny^000000~";
	mes "冰淇淋~";
	mes "来买冰淇淋！";
	next;
	switch(select("给我一个！:冰淇淋是什么？:取消。")) {
	case 1:
		mes "[冰淇淋商人]";
		mes "想要凉爽冰淇淋的人很多，所以一次最多只能订购 5 个。";
		mes "那么你要多少？";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[冰淇淋商人]";
				mes "一个都不要？";
				mes "那就让开，我还要招待其他客人。";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[冰淇淋商人]";
				mes "哇。";
				mes "你点得太多了。";
				mes "吃超过 5 个的话，你可能要和肚子里的怪物战斗。冷静点，伙计。";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[冰淇淋商人]";
			mes "伙计~！你的钱不够。";
			mes "真的只要^3355FF100 Zeny^000000~！";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[冰淇淋商人]";
			mes "你携带的东西太多了。";
			mes "购买前请减轻负重。";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[冰淇淋商人]";
		mes "‘冰淇淋是……";
		mes "等等，你不知道";
		mes "冰淇淋是什么吗？";
		mes "你到底";
		mes "住在哪块石头下面？";
		next;
		mes "[冰淇淋商人]";
		mes "我都不知道该从何吐槽这种奇怪说法。";
		mes "总之，趁冰淇淋又凉又好吃，快来买吧。";
		close;
	case 3:
		mes "[冰淇淋商人]";
		mes "别错过吃遍全大陆最好冰淇淋的机会~！";
		close;
	}
}
