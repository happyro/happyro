		mes "[Hugin's Butler]";

		close;

		close;
	}

		mes "[Hugin's Butler]";
		mes "想回来时随时来吧。";
		close;
	}

	mes "确定要购买时光 ^FF0000" + .@type$[.@s] + "^000000 靴吗？交易不可退款。";

		mes "[Hugin's Butler]";
		mes "好的，确定后再回来。";
		close;
	}
	mes "[Hugin's Butler]";
	mes "靴子已按您的要求升级。";
	if (getequipid(EQI_SHOES) == 2499) {
		delequip EQI_SHOES;
		delitem 6607,5;// Temporal_Crystal
		getitem .@equip_type[.@s],1;
	}
	close;

			close;

				mes "好的，准备好后再来。";
				close;
			}

			.@string$ = "enchant number ^6300001^000000.";
		}

			for ( .@enchant_num = 1; .@enchant_num < 5; .@enchant_num++ ) {
				for ( .@enchant_type = 0; .@enchant_type < 6 && .@card[3] != getd( ".@enchant_" + .@enchant_num + "[" + .@enchant_type + "]" ); .@enchant_type++ )
					continue;
				if (.@enchant_type < 6)
					break;
			}
			if (.@enchant_num == 5) {
				mes "[Hugin's Magician]";
				mes "发生了错误。";
				close;
			}
			.@cost = .@enchant_cost[.@enchant_num];

			mes "[Hugin's Magician]";
			if (.@enchant_num == 4) {
				.@card[2] = callfunc("F_Rand",4875,4876,4877,4878,4879,4880);// Bear's_Power, Runaway_Magic, Speed_Of_Light, Muscle_Fool, Hawkeye, Lucky_Day

			}
			else {
				.@number = .@enchant_num + 1;
				.@card[3] = getd( ".@enchant_" + (.@enchant_num+1) + "[" + .@enchant_type + "]" );
				.@string$ = "enchant number ^630000" + .@number + "^000000.";
				mes "将 ^0000FF" + .@equip_name$ + "^000000 的第 4 个插槽附魔为 ^0000FF" + .@number + "^000000 级效果。需要 ^0000FF" + .@cost + "^000000 个凝结魔法物品。";
			}
			next;
			if (select("退出","升级效果！") == 1) {
				mes "[Hugin's Magician]";
				mes "好的，准备好后再来。";
				close;
			}
		}
		if (countitem(6608) < .@cost) {
			mes "[Hugin's Magician]";
			mes "嗯，你还缺少 " + (.@cost - countitem(6608)) + " 个凝结魔法物品。再去准备一些，然后我们再谈附魔。";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		delitem 6608,.@cost;// Coagulated_Spell

		// anti-hack
		if (callfunc("F_IsEquipIDHack", EQI_SHOES, .@equip_id) || callfunc("F_IsEquipCardHack", EQI_SHOES, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]) || callfunc("F_IsEquipRefineHack", EQI_SHOES, .@equip_refine))
			close;

		delequip EQI_SHOES;
		mes "[Hugin's Magician]";
		mes "正在尝试" + .@string$;
		getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,.@card[2],.@card[3];
		close;
	default:
		mes "[Hugin's Magician]";
		mes "这不是适合此次附魔的物品。";
		mes "记住，只有 6 种时光靴可以附魔。";
		close;
	}
}

		for ( .@num = 1; .@num < 5; .@num++ ) {
			for ( .@type = 0; .@type < 6 && .@card[3] != getd( ".@enchant_" + .@num + "[" + .@type + "]" ); .@type++ )
				continue;
			if (.@type < 6)
				break;
		}
		if (.@num == 5) {
			mes "[Dark magic master]";
			mes "发生了错误。";
			close;
		}
		.@amount_coag = .@en_amount_coag[.@num];
		.@amount_poll = .@en_amount_polluted[.@num];

		mes "[Dark magic master]";
		if (.@num == 4) {
			.@card[2] = callfunc("F_Rand",4875,4876,4877,4878,4879,4880);// Bear's_Power, Runaway_Magic, Speed_Of_Light, Muscle_Fool, Hawkeye, Lucky_Day
			.@string$ = "^990000 Bonus effect ^000000 upgrade.";
		mes "要为第 3 个插槽添加随机额外效果吗？需要 ^0000ff" + .@amount_poll + " 个" + getitemname(.@polluted_id) + "、" + .@amount_coag + " 个" + getitemname(.@coagulated_id) + "以及 " + callfunc("F_InsertComma",.@cost_zeny) + " Zeny^000000。但失败风险很高……";
		}
		else {
			.@level = .@num + 1;
			.@card[3] = getd( ".@enchant_" + (.@num+1) + "[" + .@type + "]" );
			.@string$ = "enchant number ^990000" + .@level + "^000000.";
		mes "将 ^0000ff" + .@equip_name$ + "（带插槽）^000000 的第 4 个插槽附魔为 ^0000ff" + .@level + "^000000 级效果。需要 ^0000ff" + .@amount_poll + " 个" + getitemname(.@polluted_id) + "、" + .@amount_coag + " 个" + getitemname(.@coagulated_id) + "以及 " + callfunc("F_InsertComma",.@cost_zeny) + " Zeny^000000。存在失败风险……";
		}
		next;
		if (select( "退出", "升级效果！" ) == 1) {
			mes "[Dark magic master]";
			mes "好的，准备好后再来。";
			close;
		}
	}
	if (countitem(.@coagulated_id) < .@amount_coag) {
		mes "[Dark magic master]";
		mes "嗯，你还缺少 " + (.@amount_coag - countitem(.@coagulated_id)) + " 个" + getitemname(.@coagulated_id) + "。再去准备一些，然后我们再谈附魔。";
		close;
	}
	if (countitem(.@polluted_id) < .@amount_poll) {
		mes "[Dark magic master]";
		mes "嗯，你还缺少 " + (.@amount_poll - countitem(.@polluted_id)) + " 个" + getitemname(.@polluted_id) + "。再去准备一些，然后我们再谈附魔。";
		close;
	}
	if (Zeny < .@cost_zeny) {
		mes "[Dark magic master]";
		mes "你必须带来 ^0000ff" + callfunc("F_InsertComma",.@cost_zeny) + " Zeny^000000。你知道天下没有免费的东西……";
		close;
	}
	delitem .@coagulated_id, .@amount_coag;
	delitem .@polluted_id, .@amount_poll;
	Zeny -= .@cost_zeny;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", EQI_SHOES, .@equip_id) || callfunc("F_IsEquipCardHack", EQI_SHOES, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]) || callfunc("F_IsEquipRefineHack", EQI_SHOES, .@equip_refine))
		close;

	delequip EQI_SHOES;
	if (.@chances[.@num] < rand(1,100)) {
		specialeffect2 EF_LORD;
		mes "[Dark magic master]";
		mes "啊啊啊，失败了。下次好运。";
		close;
	}
	specialeffect2 EF_REPAIRWEAPON;
	mes "[Hugin's Magician]";
	mes "正在尝试" + .@string$;
	getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,.@card[2],.@card[3];
	close;
}
