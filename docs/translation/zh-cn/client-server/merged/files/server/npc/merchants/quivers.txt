payon_in01,5,134,5	script	Inventor Jaax	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[Inventor Jaax]";
	mes "嘿，你携带的东西";
	mes "太多了。为什么不先存进卡普拉仓库？存好后我们再谈，好吗？";
		close;
	}
	mes "[Inventor Jaax]";
	mes "我叫 Jaax。";
	mes "不夸张地说，";
	mes "我或许是这个时代";
	mes "^663300最伟大的发明家^000000。";
	next;
	mes "[Inventor Jaax]";
	mes "这次，我";
	mes "创造了真正";
	mes "非凡的东西。我称它们为……^663300魔法箭袋^000000！！这将作为";
	mes "箭矢革命载入史册！";
	emotion ET_BEST;
	next;
	mes "[Inventor Jaax]";
	mes "我研究魔法和箭袋多年，日夜工作，终于想出用魔法压缩箭矢的方法！有了魔法箭袋，你能携带更多箭矢，却只需承受更少重量！";
	next;
	mes "[Inventor Jaax]";
	mes "想试试我的箭袋吗？我相信像你这样的人一定能欣赏我的天才之处！";
	next;
	switch(select("箭袋:铁箭袋:钢铁箭袋:秘银箭袋:火箭袋:银箭袋:风箭袋:石箭袋:水晶箭袋:暗影箭袋:无形箭袋:生锈箭袋")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Is there";
	//mes "anything you want?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[Inventor Jaax]";
		mes "太棒了！";
		mes "你带着箭矢吗？我可以提供一个箭袋，装入你的 "+ getitemname(.@item_id) +" "+ .@amount +" 支，只需 ^FF3131"+ .@zeny_req +" Zeny^000000。";
		next;
		switch(select("尽可能将箭矢装入箭袋:购买 1 个箭袋:取消")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "箭矢数量：";
			mes "^3131FF"+.@arrows+" 支^000000";
			mes "最多可购买的";
			mes "箭袋数量：";
			mes "^3131FF"+.@quiver+" 个^000000";
			mes "所需 Zeny：^3131FF"+.@arrow_zeny01+"^000000";
			next;
			mes "[Inventor Jaax]";
			mes "你想用当前携带的箭矢";
			mes "尽可能购买";
			mes "多的箭袋吗？";
			mes "";
			next;
			if (select("是:取消") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[Inventor Jaax]";
					mes "给你！";
					mes "记住，^FF0000当携带重量达到最大负重的 90% 时，你将无法使用箭袋^000000。";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[Inventor Jaax]";
					mes "所以……";
					mes "记得时常留意";
					mes "自己携带了多少东西，";
					mes "这样就不会有问题。";
					mes "";
					mes "";
					close;
				}
				else {
					mes "[Inventor Jaax]";
				mes "抱歉，你没有足够的 Zeny。为了开发这项革命性技术，我辛苦了多年，";
				mes "不能就这样免费送给你！";
					close;
				}
			}
			mes "[Inventor Jaax]";
			mes "你改变主意了？";
			mes "拥有箭袋的荣耀";
			mes "近在眼前啊？";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[Inventor Jaax]";
				mes "给你！";
				mes "记住，^FF0000当携带重量达到最大负重的 90% 时，你将无法使用箭袋^000000。";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[Inventor Jaax]";
				mes "所以……";
				mes "记得时常留意";
				mes "自己携带了多少东西，";
				mes "这样就不会有问题。";
				mes "";
				mes "";
				close;
			}
			else {
				mes "[Inventor Jaax]";
				mes "你连 "+ .@zeny_req +" Zeny 都没有？";
				mes "真抱歉。我没想到你竟然";
				mes "如此……";
				mes "^333333一贫如洗^000000。";
				mes "";
				close;
			}
		case 3:
			mes "[Inventor Jaax]";
			mes "什么……？";
			mes "难道你没看出这项发明能永远改变携带箭矢的方式吗？！未来就是现在！";
			close;
		}
	}
	else {
		mes "[Inventor Jaax]";
		mes "这个箭袋最多能装 "+ .@amount +" 支箭矢。这是用我的秘密方法制成的，";
		mes "因此箭矢和箭袋的总重量比单独携带箭矢还轻。";
		next;
		mes "[Inventor Jaax]";
	mes "这是科学的奇迹！只要带来至少 "+ .@amount +" 支箭矢，以及每个箭袋所需的 "+ .@zeny_req +" Zeny，你就能亲自体验。";
		close;
	}
}
