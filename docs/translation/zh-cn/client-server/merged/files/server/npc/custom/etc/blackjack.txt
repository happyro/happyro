//= rAthena Project
//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	Black Jack Dealer	57,{
	mes "[Dealer]";
	mes "你好！想玩一局二十一点吗？";
M_Menu:
	next;
	menu "What are the rules?",M_0, "I want to play.",L_Play, "No thanks.",M_End;

M_0:
	mes "[Dealer]";
	mes "你想了解什么？";
	mes "我会为你说明规则。";
	mes "请仔细听。";
	mes "这样就能开始游戏。";
sM_Menu0:
	next;
	menu "The basics.",sM_0a, "Winning and loosing.",sM_0b, "'Double Down'.",sM_0c,
	"Ace value.",sM_0d, "Nothing.",M_Menu;

sM_0a:
	mes "[Dealer]";
	mes "二十一点是一种纸牌游戏，目标是得到 21 点，";
	mes "或者在不超过 21 点的前提下尽可能接近 21 点。";
	next;
	mes "[Dealer]";
	mes "游戏开始前，玩家必须下注。下注后，庄家和玩家各发 2 张牌。";
	mes "根据手中的牌，你可以选择^5533FF‘停牌’^000000，";
	mes "也可以选择^5533FF‘要牌’^000000。";
	mes "请谨慎做出决定。";
	next;
	mes "[Dealer]";
	mes "选择‘停牌’就是告诉庄家你不需要更多牌。";
	mes "庄家可以根据需要继续要牌。";
	mes "停牌时你必须至少有 ^FF3355 13 点^000000。";
	mes "庄家只有达到 ^FF3355 17 点或以上^000000 才能停牌。";
	next;
	mes "[Dealer]";
	mes "选择‘要牌’就是告诉庄家你想再要一张牌。";
	mes "要更多牌可以增加你的总点数。";
	mes "只要你的点数 ^FF3355低于 21^000000，";
	mes "就可以从牌堆中再抽牌。";
	goto sM_Menu0;
sM_0b:
	mes "[Dealer]";
	mes "二十一点有 3 种获胜或失败的情况。";
	next;
	mes "[Dealer]";
	mes "1）回合结束时，如果你的点数高于庄家，就赢得回合；";
	mes "如果点数低于庄家，就会输掉回合。";
	next;
	mes "[Dealer]";
	mes "2）如果要牌后总点数超过 21，就会自动输掉回合。";
	mes "这称为^5533FF‘爆牌’^000000。若庄家爆牌，你将赢得回合。";
	next;
	mes "[Dealer]";
	mes "3）如果前两张牌的总点数为 21，就会自动赢得回合。";
	mes "这称为^5533FF‘二十一点’^000000，通常由一张 A 和一张 10 点牌组成。";
	mes "如果庄家拿到二十一点，庄家会自动获胜。";
	next;
	mes "[Dealer]";
	mes "4）除了胜负，你也可能和庄家平局。";
	mes "如果回合结束时你和庄家的点数相同，";
	mes "就会平局，没有赢家，也不会损失或赢得金钱。";
	mes "下注将被退还。";
	next;
	mes "[Dealer]";
	mes "这称为与庄家^5533FF‘和局’^000000。";
	mes "你和庄家同时拿到二十一点时也适用。";
	mes "请记住这个规则。";
	goto sM_Menu0;
sM_0c:
	mes "[Dealer]";
	mes "‘加倍下注’可以将当前赌注翻倍，";
	mes "但代价是只能再要一张牌。";
	mes "这个选项只在每回合开始时可用。";
	mes "请谨慎选择。";
	next;
	mes "[Dealer]";
	mes "适合加倍下注的例子是：";
	mes "前两张牌正好得到 11 点。";
	mes "下一张牌很可能让你得到 21 或 20 点。";
	mes "这种牌面很适合加倍。";
	goto sM_Menu0;
sM_0d:
	mes "[Dealer]";
	mes "‘A’牌在二十一点中很特殊，因为它有两种点数。";
	mes "A 可以算作 11 点，也可以算作 1 点。";
	mes "例如 A 和 4 可以得到 15 点或 5 点。";
	mes "系统会根据情况自动计算。";
	next;
	mes "[Dealer]";
	mes "如果你选择停牌，A 会自动计算为";
	mes "11点，因此总点数为15点。";
	next;
	mes "[Dealer]";
	mes "如果你选择要牌并得到 9，A 会自动算作 1 点，";
	mes "这样总点数就是 14 点。";
	mes "如果 A 算作 11 点，总点数会超过 21，";
	mes "你就会输掉回合。";
	next;
	mes "[Dealer]";
	mes "正是因为 A 的灵活性，";
	mes "它才成为游戏中最强的牌。";
	goto sM_Menu0;
M_End:
	mes "[Dealer]";
	mes "欢迎随时回来";
	close;

//================
L_Play:
	mes "[Dealer]";
	mes "请下注……";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Too rich for my blood....",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[Dealer]";
	mes "抱歉，你没有足够的 Zeny 来下注。";
	close;

L_Cont0:
	mes "（正在发牌）";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- 这是^FF5533庄家的^000000牌：";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    庄家的点数为：^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- 这是^5533FF你的^000000牌：";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    YOU have: ^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    You have: ^5533FF" +@playTot+ "^000000, or ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay, "Double Down",M_Double;
	if(@dealerTurn == 0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay;

M_Stay:
	mes "[Dealer]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "庄家停牌。";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "庄家要牌";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "抱歉，你的总点数不够停牌。必须要牌。";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[Dealer]";
	mes "玩家选择了加倍下注。当前赌注将翻倍，";
	mes "并且只能再要 1 张牌。";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[Dealer]";
	mes "恭喜你，赢得了比赛！";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[Dealer]";
	mes "很遗憾，你输了。";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[Dealer]";
	mes "这是和局。你和庄家点数相同。";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
