xmas_in,100,96,4	script	Father Christmas::Santa2	718,{
	mes "[Santa Claus]";
	if(xmas_npc==0) set xmas_npc, 1; // For Lutie & Bard quest
	if(#event_xmas > 0 && #event_xmas < 30 ) goto L_Start;
	mes "圣诞快乐！";
	if(Class==0 || #event_xmas>=30 ) close; //anti exploit protection
	mes "我有一份礼物给你！呵呵呵！";
	getitem rand(664,667),1; //gives one of 4 gift boxes

		next;
		mes "[Santa Claus]";
		mes "像传说中的格林奇一样，他拿走孩子们的所有玩具，据为己有。";
		next;
		mes "[Santa Claus]";
		mes "我太忙了，要为明年制作玩具，无法亲自出去找他。";
		mes "所以请你出去替我消灭这个人。";
		next;
		mes "[Santa Claus]";
		mes "不过他手里有我的魔法袋，每次你攻击他，他都会钻进袋子逃到别处。";
		next;
		mes "[Santa Claus]";
		mes "不过他匆忙时经常掉东西。";
		mes "如果他掉下一只用来偷孩子玩具的破袜子，请把它捡起来。";
		next;
		mes "[Santa Claus]";
		mes "收集 3 只袜子，我就给你一只实验室里的神秘箱原型。";
		mes "它会随机吐出礼物，能省我很多工作。";
		close;
		M_1:
		mes "[Santa Claus]";
		if(countitem(7034) < 3) goto L_NotEnuf;
		delitem 7034,3;
		mes "看来你干掉那些假圣诞老人做得很好。继续努力！";
		next;
		getitem 644,1; //Gift Box Prototype
		set #event_xmas,#event_xmas+1;
		mes "[Santa Claus]";
		mes "这是你的奖励。";
		mes "再收集 3 只，我会再给你一个。";
		mes "希望你能得到好东西。";
		close;
		
		L_NotEnuf:
	mes "你没有足够的袜子作为证据。";
		mes "去消灭那些邪恶的圣诞老人，收集更多，我会奖励你。";
		close;
		M_End:
		mes "[Santa Claus]";
		mes "我明白了。至少我们会在圣诞早晨再见。";
		close;
		
		OnInit:
		disablenpc "Santa Claus";
		end;
