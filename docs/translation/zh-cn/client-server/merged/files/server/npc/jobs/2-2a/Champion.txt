valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Champion]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[Champion]";
		mes "技能。";
		mes "速度。";
		mes "力量。";
		mes "敏捷。";
		next;
		mes "[Champion]";
		mes "武僧宗师可以";
		mes "从这些方面获益。";
		mes "但是，一个人一生";
		mes "只能掌握这么多……";
		mes "";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[Champion]";
		mes "时机到了。";
		mes "伟大的英雄们该站出来，";
		mes "对抗肆虐";
		mes "米德加尔特世界的";
		mes "邪恶势力了！";
		next;
		mes "[Champion]";
		mes "您愿意";
		mes "成为武僧宗师吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[Champion]";
			mes "准备好后，";
			mes "随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[Champion]";
			mes "您仍然可以学习更多技能。请在回来找我之前";
			mes "用完所有剩余技能点。";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[Champion]";
		mes "恭喜！";
		mes "作为武僧宗师生活，";
		mes "用双拳的力量";
		mes "为这个世界";
		mes "带来光明。";
		close;
	}
	mes "[Champion]";
	mes "欢迎来到";
	mes "英灵殿，";
	mes "荣耀之厅。";
	next;
	mes "[Champion]";
	mes "请在这里";
	mes "好好休息。";
	mes "向勇士致敬！";
	close;
	}
