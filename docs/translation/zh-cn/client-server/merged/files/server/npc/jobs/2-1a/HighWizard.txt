valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[高阶巫师]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[高阶巫师]";
		mes "我们高阶巫师肩负着";
		mes "责任，";
		mes "使用毁灭性魔法";
		mes "实现正当目的。";
		next;
		mes "[高阶巫师]";
		mes "成为高阶巫师前需要";
		mes "一生的训练。";
		mes "你能想象吗，";
		mes "如果我们的力量";
		mes "落入坏人手中会发生什么？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[高阶巫师]";
		mes "是时候了。";
		mes "米德加尔特需要";
		mes "能够驾驭";
		mes "最强魔法的人……";
		next;
		mes "[高阶巫师]";
		mes "你想成为";
		mes "高阶巫师吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[高阶巫师]";
			mes "准备好后，";
			mes "随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[高阶巫师]";
			mes "你仍然可以学习更多技能。请在回来前使用";
			mes "所有剩余技能点。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[高阶巫师]";
		mes "恭喜。";
		mes "作为高阶巫师，";
		mes "我希望你使用";
		mes "力量为受压迫者带来";
		mes "和平。";
		close;
	}
	mes "[高阶巫师]";
	mes "欢迎";
	mes "来到英灵殿，";
	mes "荣誉之厅。";
	next;
	mes "[高阶巫师]";
	mes "请让自己";
	mes "舒适地";
	mes "待在这里。";
	mes "向勇士致敬！";
	close;
}
