		close2;

		end;

			mes "请注意，战场奖励物品是 ^FF0000角色绑定^000000 的。确定要兑换吗？";
			next;
		}

		mes "[艾伦德克]";
		if (.@j == 3) {
			mes "你取消了兑换。";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "感谢你的兑换。";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "你的"+getitemname(.@cost)+"不足。";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[艾克斯尔·罗斯]";
			mes "吉约姆军万岁！";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "克鲁瓦军万岁！";
		}
		mes "让我们享受光荣的胜利吧！";
		mes strcharinfo(0)+"，这是胜利的象征。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[艾克斯尔·罗斯]";
			mes "你输了，但你为这场战斗付出了努力。";
			mes "这是吉约姆·马洛尔对你坚定不移的奖励！";
			mes "把这次失败当作教训，下次你一定会赢。";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "哦，"+strcharinfo(0)+"，别难过。";
			mes "虽然我们没有获胜，但我们已经尽力了。";
			mes "这是克鲁瓦王室的礼物，请不要忘记这场战斗。下一场我们会赢。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
