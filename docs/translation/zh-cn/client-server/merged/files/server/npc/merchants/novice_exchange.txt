	function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	
	if (countitem(.@item_req) < .@req_amount) {
	mes "[Merchant]";
	mes "嘿，你答应的";
	mes "那些"+ getitemname(.@item_req) +"在哪里？";
	mes "把"+ getitemname(.@item_req) +"给我！";
	close;
	}
		else {
		mes "[Merchant]";
		mes "好，让我检查一下";
		mes "看看你身上有多少"+ getitemname(.@item_req) +"。嗯……";
		next;
		mes "[Merchant]";
		mes "你共有"+ countitem(.@item_req) +" 个"+ getitemname(.@item_req) +"。";
		mes "我可以给你";
		mes (countitem(.@item_req)/.@req_amount) +" 瓶红色药水。";
		next;
		mes "[Merchant]";
		mes "你觉得怎么样？";
			mes "成交吗？";
			next;
			if (select("成交。:不成交。") == 1) {
			mes "[Merchant]";
		mes "你知道自己想从";
	mes "这些"+ getitemname(.@item_req) +"中换取";
		mes "多少瓶红色药水，对吧？";
		next;
		mes "[Merchant]";
		mes "现在一次最少可以交换 1 瓶";
		mes "最多 100 瓶红色药水。";
		mes "改变主意的话输入‘0’取消。";
	next;
		while(1) {
		input .@input,0,101;
		if (.@input == 0) {
		mes "[Merchant]";
		mes "什么……？";
		mes "你干吗还来找我？";
	mes "真是优柔寡断，对吧？";
close;
}
else if (.@input > 100) {
	mes "[Merchant]";
		mes "一次不能给你超过 100 瓶";
		mes "红色药水。我们再试一次。";
		next;
		}
		else {
		break;
		}
	}
	.@put_out_item = .@input * .@req_amount;
	if (countitem(.@item_req) < .@put_out_item) {
	mes "[Merchant]";
	mes "糟糕，你输入的数量不对。请重新确认可以交换的红色药水数量。";
	close;
	}
	mes "[Merchant]";
	mes "给你！";
	mes "检查一下我给你的红色药水数量，应该没问题。谢谢，这笔交易不错~";
	delitem .@item_req,.@put_out_item; //Shell
	getitem 501,.@input; //Red_Potion
	close;
	}
		mes "[Merchant]";
		mes "嗯~";
		mes "好吧。";
		mes "不过，红色药水对你这样的";
		mes "冒险者不是更有用吗？";
		close;
		}
		}
		
		//============================================================ 
			// Old changelog
			//============================================================ 
			//= 1.1 Made it as a function. U can add your own Merchs [Lupus]
			//= 1.1a Fixed item select [KarLaeda]
		//============================================================
	mes "如果改变主意";
		mes "再回来。";
		close;
		}
		}
