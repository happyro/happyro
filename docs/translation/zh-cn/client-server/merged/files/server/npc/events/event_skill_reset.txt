//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hypnotist Teacher	124,{
	set @npcname$,"[^D5A500Hypnotist^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "我已经说过，这个活动只能完成一次。";
		close;
	}
	if(BaseLevel < 60) {
		mes "基础等级达到 60 级或更高后再回来。";
		close;
	}
	if(SkillPoint != 0) {
		mes "想让我继续，你必须用完所有技能点。";
		mes "请尽快再来！";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "想重置技能吗？";
		mes "很抱歉，但是……";
		next;
		mes @npcname$;
		if(Weight) {
			mes "携带物品时";
			mes "无法重置技能。";
			} else if(checkcart())
		mes "请放下手推车，我们再继续。";
			else if(checkfalcon())
		mes "请让猎鹰归队，我们再继续。";
			else if(checkriding())
		mes "请解除大嘴鸟骑乘，我们再继续。";
			next;
		mes @npcname$;
		mes "请尽快回来！";
		close;
		}
	mes "你好，我是技能重置师。";
	mes "你的名字是 ^0080FF"+strcharinfo(0)+"^000000。";
	mes "我能帮你什么？";
	next;
	menu "^009500Information about Reset skills.^000000",L_Info,"^00B6FFReset skills.^000000",L_Reset,"^000088Nevermind^000000",-;
	
mes @npcname$;
	mes "如果你想重置技能，";
	mes "知道在哪里能找到我吧！！"; 
	close;
	
L_Info:
mes @npcname$;
	mes "技能重置不是免费的！！";
	mes "技能重置费用为 ^D5A50020000 Zeny × 基础等级^000000。";
	mes "没错……每级基础等级需要 20000 Zeny。";
	next;
	mes @npcname$;
	mes "对了，还有一件事！";
	mes "重置技能时，已装备的手推车、猎鹰或大嘴鸟";
	mes "都会被解除。";
	next;
	mes @npcname$;
	mes "只能再次重置一次技能点。";
	mes "今后请谨慎分配技能。";
	close;
	
	L_Reset:
mes @npcname$;
	mes "开始技能重置前，";
	mes "请先告诉我你的基础等级。"; 
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "我的基础等级是 ^AA00AA等级 " + BaseLevel + " 基础等级。^000000"; 
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "技能重置服务总费用为 ^529DFF" + @zeny + " Zeny^000000。";
	next;
	if(Zeny < @zeny) {
	mes @npcname$;
		mes "看来你的钱不够。";
		mes "有机会再来吧。";
		emotion ET_SCRATCH;
		close;
		}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "谢谢。";
	emotion ET_THANKS;
	//	logmes "SKILL RESET EVENT";
close;
	}
