function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "你的"+getitemname($@wed_ring)+"怎么了？你没弄丢吧……？我们需要它才能继续仪式！";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "你似乎没有足够空间携带戒指……请腾出空间后再回来领取伴侣的戒指。";
		close;
	}
	mes "这是给你新娘的婚戒。";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "你的新娘领取戒指后，仪式就会开始。";
		close;
	}
}

function SF_StartCeremony {
	mes "现在开始婚礼仪式，并宣布你们成为夫妻。";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes $@msg$+"在哪里？少了一个人，我无法为你们举行婚礼……";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "孩子，你把你的"+getitemname(7170)+"弄到哪里去了？";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "孩子，仪式期间你必须一直穿着"+getitemname(2338)+"……";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "孩子，现在还不能脱下你的"+getitemname(2206)+"……";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "好的，你想登记为哪一方？";
		next;
		set @submenu, select("新郎","新娘","取消");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "好的，你要登记为新郎吗？";
		next;
		if (select("是","我改变主意了。")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "好的，你要登记为新娘吗？";
		next;
		if (select("是","我改变主意了。")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你现在登记为新郎。";
		mes "请尽快让新娘登记。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你现在登记为新娘。";
		mes "请尽快让新郎登记。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "准备好后再回来。";
		close;
	}
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "新郎";
	if (@bride)
		set @type$, "新娘";
		
	mes "["+@name$+"]";
	mes "登记为"+@type$+"之前，先确认你满足所有条件……";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "哎呀，你还太年轻，不能考虑结婚！";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "想结婚就必须穿着"+getitemname(@item)+"。";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "你的"+getitemname(2206)+"在哪里？这是礼服的必要搭配。";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "戒指呢？交换婚戒需要一个"+getitemname($@wed_ring)+"，亲爱的。";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "抱歉，你没有足够的钱支付登记费。";
		mes "收集到"+@cost+" Zeny 后再回来。"; 
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
