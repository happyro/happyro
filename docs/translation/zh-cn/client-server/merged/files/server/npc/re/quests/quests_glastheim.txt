glast_01,195,273,3	script	Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[Hugin's Follower]";
	mes "我不像主人 Hugin 那样能穿越时间和次元，但可以假装做到。";
	next;
	mes "[Hugin's Follower]";
	mes "是什么让这座古拉斯特海姆充满如此强大的黑暗力量？";
	next;
	if( select( "我不在乎:真有趣，请继续说" ) == 2 ){
		mes "[Hugin's Follower]";
		mes "多亏 Hugin，我找到了进入旧古拉斯特海姆的方法，那里充满强大的黑暗力量。";
		next;
		mes "[Hugin's Follower]";
		mes "怎么样？想去探索那里吗？";
		next;
		if( select( "我不想探索:我真的想去！" ) == 2 ){
			mes "[Hugin's Follower]";
			mes "当然可以，但这不是免费服务！我需要收取一些报酬。呵呵……";
			next;
			mes "[Hugin's Follower]";
			mes "我可以送您去两个地方。您想去哪里？";
			next;
			switch( select( "取消:修道院墓地（等级 100～130）- 5000z:二楼走廊（等级 140 以上）- 10000z" ) ){
				case 1:
					mes "[Hugin's Follower]";
					mes "好的，改变主意后随时回来。我想您会有很多话想和我说……";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[Hugin's Follower]";
				mes "什么？您以为这项服务免费吗？";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[Hugin's Follower]";
	mes "好吧，如果您没兴趣，那就算了。";
	close;
}
