//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我是武器工匠。";
	mes "我可以精炼各种武器、防具和装备，";
	mes "告诉我你想精炼什么吧。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想你拥有的物品中没有可以精炼的……";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你那里没有穿戴";
		mes "任何我可以精炼的";
		mes "物品。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想这件物品";
		mes "完全无法精炼……";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我不能再精炼这件物品了。";
		mes "它已经达到";
		mes "精炼上限！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "精炼需要";
		mes "一个^003366"+getitemname(.@material)+"^000000，";
		mes "以及 " + .@price + " Zeny 的服务费。";
		mes "你真的要继续吗？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "好吧……";
			mes "不用着急。";
			mes "慢慢来。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "糟糕！如果继续精炼，";
			mes "这件物品有可能会";
			switch(.@material) {
			case 985:
				mes "被破坏！这意味着^FF0000这件装备^000000以及附在防具上的^FF0000卡片^000000或特殊属性都会^FF0000消失^000000。";
				break;
			default:
				mes "被破坏，你会^FF0000失去武器^000000、武器上的^FF0000卡片^000000，";
				mes "以及附加的特殊属性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我已经说得很清楚了。";
			mes "武器一旦被破坏，";
			mes "就再也无法找回。";
			mes "你真的可能会永远";
			mes "^FF0000失去这件武器^000000。";
			mes "你还要继续精炼吗？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意……";
				mes "我也许是出色的精炼师，但有时也会犯错。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "看来你的 Zeny 或 "+getitemname(.@material)+"";
			mes "不够……再去准备一些吧。";
			mes "你需要我的话，我会一直";
			mes "在这里。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你以为我是傻瓜吗？！";
			mes "你趁我没注意时换了物品！滚出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "哦！我的天！";
				mes "该死！又来了！";
				mes "非常抱歉，不过熟能生巧嘛。";
				mes "嗯，对吧？嘿嘿……";
			} else if(.@lose == 2) {
				mes "不——！！！";
				mes "它坏掉了！";
				mes "对、对不起！";
			} else {
				mes "糟糕！";
				mes "它承受不了";
				mes "更多的精炼了！";
				mes "真是抱歉……";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "嘿嘿！";
			mes "又一次，";
			mes "大师的作品";
			mes "毫无瑕疵～";
		} else if(.@win == 2) {
			mes "成功……！";
			mes "今天我的惊人";
			mes "才能再次大放异彩，";
			mes "闪耀夺目。";
		} else {
			mes "嘿嘿！";
			mes "全部完成了。";
			mes "毫无疑问，我的作品";
			mes "一定令你满意。";
			mes "纯粹而彻底的完美～";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以将它精炼到安全上限，或按你指定的次数精炼。由你选择。";
		next;
		.@menu2 = select("精炼到安全上限。","我来决定精炼次数。","我改变主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你想让我精炼物品多少次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "我不能将这件物品精炼那么多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "这将尝试把装备精炼到超过安全上限 " + .@refinecheck + " 次。装备可能会被破坏……你确定吗？";
			next;
			if(select("是……","不……") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "这是你说的……那就这样吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "这是你说的……那就这样吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "这需要 " + .@refinecnt + " 个" + getitemname(.@material) + "和 " + .@fullprice + " Zeny。你确定吗？";
	next;
	if(select("是","不……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "这是你说的……那就这样吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "你只有这些吗？很遗憾，我不能以更低的价格为你工作。换位思考一下吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "你看……你身上没有装备……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "当啷……你以为我会蠢到没发现吗？！";
			mes "你换了装备……";
			mes "在我用锤子把你击晕前滚出去！！";
			close;
		} 
		mes "当啷，当啷！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊啊啊！！！非常抱歉……我警告过你可能会这样……";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "这是剩下的 Zeny 和材料……";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "全部完成了……欢迎下次再来。";
	close;
}

function	script	phramain	{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 你目前携带了 -";
		mes "- 太多物品。-";
		mes "- 请重试 -";
		mes "- 清理背包后。-";
		close;
	}
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "我出售两种";
	mes "用于淬炼武器的金属。";
	mes "我有供 1 级武器使用的 ^007777普拉康^000000，";
	mes "以及供 2 级武器使用的 ^007777艾姆威泰勒矿石^000000";
	mes "。";
	next;
	switch(select("普拉康 - 200 Zeny:艾姆威泰勒矿石 - 1000 Zeny:询问其他金属")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "其他金属？";
		mes "高等级武器或任何护甲都需要特殊金属。不过，神之金属和铝原石很";
		mes "难找到……";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "你想买多少？";
	mes "不想买请输入数字“0”。";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "交易已";
			mes "取消。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "好，你最多可以";
			mes "购买 500 个。";
			mes "不能更多，";
			mes "明白了吗？很好。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "呃……";
		mes "你没有足够 Zeny 购买";
		mes "";
		mes ""+ .@input +" 个。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";
		mes "背包空间不足，无法给你物品。把多余物品存入卡普拉仓库后重试吧？";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "给你！";
	mes "感谢";
	mes "惠顾。";
	close;
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "你好！";
	mes "想让我";
	mes "修理什么物品吗？";
	mes "物品修理";
	mes "交给我就行！";
	next;
	switch(select("我确实有些物品……:现在没有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哇，真不可思议！";
			mes "你一定把物品保管得很好，没有一件物品损坏！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "如果大家都像你一样，我就要失业了！！哈哈~！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";
		mes "让我看看……";
		mes "在你的所有物品中，";
		mes "共有 " + .@checkitem + " 件物品损坏。";
		mes "要修理吗？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每件物品修理费为 " + .@repairprice + " Zeny。修理全部损坏物品需要 " + .@totalcost + " Zeny！要修理这些物品吗？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "哎呀哎呀……";
				mes "收到修理账单前先看看钱包吧！你的 Zeny 不够，我什么都修不了。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好了！全部完成。以后请稍微小心一点，物品也是有寿命的。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？好像有些不对。等等……装备需要修理的物品后再回来找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "这倒不是我的损失，但让物品一直损坏可不好。你应该尽快修理！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "呵呵呵……";
		mes "如果没有";
		mes "物品要修理，";
		mes "就没必要";
		mes "来找我。";
		close;
	}
}
