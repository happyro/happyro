-	script	ZondaTeleportationServices	HIDDEN_NPC,{
	cutin "zonda_do01", 2;
	mes "[Zonda 员工]";
	mes "欢迎。";
	mes "我们的团队始终努力提升服务质量，超越你的期待。";
	mes "请选择目的地。";
	next;
	switch( select( "北方彩虹桥", "伊克拉斯", "斯普兰迪德", "米德加尔德远征营地", "马努克", "卡米达尔山外围", "取消。") ) {
	case 1:
		callsub( S_Zone, "北方彩虹桥", 17000,850, 7213 ); // Teleport value not confirmed
		warp "bif_fild02",160,230; // Unverified Coordinates
		end;
	case 2:
		callsub( S_Zone, "伊克拉斯", 17000,850, 11311 );
		warp "eclage",111,39;
		end;
	case 3:
		callsub( S_Zone, "斯普兰迪德", 17000,850, 10078 );
		warp "splendide",171,115;
		end;
	case 4:
		callsub( S_Zone, "米德加尔德远征营地", 19000,950, 10078 );
		warp "mid_camp",210,288;
		end;
	case 5:
		callsub( S_Zone, "马努克", 22000,1100, 10078 );
		warp "manuk",263,177;
		end;
	case 6:
		callsub( S_Zone, "卡米达尔山外围", 24000,1200, 7184 );
		warp "dic_fild01",188,187;
		end;
	case 7:
		mes "[Zonda 员工]";
		mes "我们始终努力为客户提供最优质的服务。";
		mes "感谢你使用我们的服务。";
		close3;
	}

S_Zone:
	.@zeny_req = getarg(1);
	.@item_req = getarg(2);
	.@quest_req = getarg(3);
	
	if (isbegin_quest(.@quest_req) == 0) {
		mes "[Zonda 员工]";
		//mes getarg(0);
		mes "使用我们的传送服务前，你必须完成前往" + getarg(0) + "的进入任务。";
		close3;
	}
	mes "[Zonda 员工]";
	mes getarg(0) + "可使用 ^ff0000" + .@zeny_req + " Zeny^000000 或 ^ff0000" + .@item_req + " 份马兰岛罐装特产^000000。";
	next;
	switch( select( .@zeny_req + " Zeny", .@item_req + " 份马兰岛罐装特产", "取消" ) ) {
	case 1:
		if (Zeny < .@zeny_req)
			break;
		Zeny -= .@zeny_req;
		return;
	case 2:
		if (countitem(12636) < .@item_req)
			break;
		delitem 12636,.@item_req; // Malang_Sp_Can
		return;
	case 3:
		mes "[Zonda 员工]";
		mes "能告诉你其他服务区域吗？";
		close3;
	}
	mes "[Zonda 员工]";
	mes "先生！你没有足够的所需费用或物品。";
	mes "请下次再来。";
	close3;
}
