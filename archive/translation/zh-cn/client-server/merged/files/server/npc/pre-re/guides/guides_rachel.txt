rachel,138,146,5	script	Rachel Guide	934,{
	mes "[拉赫向导]";
	mes "欢迎来到阿鲁纳佩尔兹首都";
	mes "拉赫，女神芙蕾雅的";
	mes "温暖气息所及之地。";
	mes "如果这是你第一次使用";
	mes "向导服务，建议先查看";
	mes "不如先查看“公告”菜单？";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("村庄指南:移除小地图标记:注意事项:取消")) {
		case 1:
			mes "[拉赫向导]";
			mes "我可以告诉你拉赫任何建筑的位置。";
			mes "你想去哪里？";
			if (.@compass_check == 0) {
				mes "你想让我";
				mes "标记地点";
				mes "在小地图上吗？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("切修尔姆尼尔:旅馆:武器店:工具店:机场:取消")) {
				case 1:
					mes "[拉赫向导]";
					mes "切修尔姆尼尔是教皇、女神芙蕾雅化身居住的圣地。";
					mes "沿北方道路即可找到建筑。";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[拉赫向导]";
					mes "你可以在旅馆消除旅途疲劳。";
					mes "我旁边左侧的建筑就是拉赫旅馆。";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[拉赫向导]";
					mes "想看看拉赫出售的武器吗？";
					mes "武器店就在西门附近。";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[拉赫向导]";
					mes "拉赫工具店出售最高品质的药水。";
					mes "它就在西门附近。";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[拉赫向导]";
					mes "机场位于东门外。";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[拉赫向导]";
					mes "想移除小地图上的所有位置标记，";
					mes "只要再问我一次，然后选择“从小地图移除标记”菜单。";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[拉赫向导]";
			mes "好了，标记已移除。有其他地点要询问就告诉我。";
			break;
		case 3:
			mes "[拉赫向导]";
			mes "使用“村庄指南”菜单时，";
			mes "建筑位置会标记在屏幕右上角的小地图上。";
			mes "看不到小地图时，使用快捷键“Ctrl+Tab”或点击基本信息窗口的“地图”按钮。";
			mes "如果看不到整个村庄地图，也可以点击“-”按钮缩小小地图。";
			break;
		case 4:
			mes "[拉赫向导]";
			mes "希望你在阿鲁纳佩尔兹";
			mes "旅途愉快。";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
