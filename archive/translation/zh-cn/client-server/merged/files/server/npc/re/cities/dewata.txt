		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "交通费为 10,000 Zeny，VIP 只需 1,000 Zeny。";
	} else {
		set .@cost,10000;
	mes "交通费为 10,000 Zeny。";
	}

	switch(select("是！:不。")) {
	case 1:
		if (Zeny >= .@cost) {
			mes "[Dewata Sailor]";

			mes "祝你旅途愉快~";
			close2;
			set Zeny, Zeny - .@cost;

			end;

		mes "[Dewata Sailor]";
