moc_para01,22,16,5	script	Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- 你的物品太多了，请先腾出空间。-";
		close;
	}
	set .@ticket_cost, 200;
	mes "[目录魔法师]";
	mes "看……格芬的魔法学院";
	mes "现在正在直接出售";
	mes "全新的";
	mes "^007777万能目录银币^000000！";
	next;
	mes "他正看着手上写的文字。";
	mes "他似乎太忙了，";
	mes "根本没注意到你。";
	mes "完全没有看你一眼。";
	next;
	mes "[目录魔法师]";
	mes "全新的目录";
	mes "刚刚上市！";
	mes "你可以查看";
	mes "能兑换哪些物品，";
	mes "以及商人正在出售哪些物品。";
	mes "你可以在目录界面查看详情。";
	next;
	mes "[目录魔法师]";
	mes "^007777万能目录银币^000000";
	mes "只要 200 Zeny！";
	mes "一次最多可以买 50 个！";
	next;
	switch(select("购买 ^007777万能目录银币^000000:我不需要它！:你看起来不像商人！")) {
	case 1:
		mes "[目录魔法师]";
		mes "你……想要全部 50 个？";
		mes "还是只需要几个？";
		next;
		while (1) {
			input .@input;
			mes "[目录魔法师]";
			if (.@input == 0) {
				mes "交易已停止！";
				mes "我不知道接下来该做什么……";
				mes "我该怎么办？";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "数量应该少于 50 个。";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "你要购买的目录总数为 " + .@input + " 个。";
		mes "需要支付 " + .@sell + " Zeny。";
		if (Zeny < .@sell) {
			mes "看来你的钱不够。";
			mes "你可以出售装备";
			mes "来赚更多钱……如果你愿意。";
			close;
		}
		mes "好的，给你。";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[目录魔法师]";
		mes "啊……你真的需要这个";
		mes "物品……要我再解释一遍吗？我不知道";
		mes "接下来该做什么……";
		mes "我该怎么办？";
		mes "你真的不需要它吗？";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[目录魔法师]";
		mes "我的真正职业是魔法师。";
		mes "我本来想请商人公会帮忙销售，";
		mes "但他们竟然想把目录";
		mes "作为扫帚销售的附录。";
		mes "所以我只好自己卖。";
		mes "这真是让人头疼。";
		next;
		mes "[目录魔法师]";
		mes "这里有很多来来往往的冒险者。";
		mes "我相信很快就能卖给他们所有人。";
		mes "你不觉得吗？";
		mes "那么……开始吧！！请多买一些！拜托……！";
		mes "请支持我的生意！";
		close;
	}
}
