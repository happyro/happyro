-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "女士们先生们！！！";
	mes "你们听过传闻，却从未亲眼见过";
	mes "世界上最稀有的钻石！";
	mes "^FF82FF‘17 克拉钻石’^000000现在低价出售！！";
	mes "如果你对珍贵钻石感兴趣，就听好了！";
	next;
	
	mes .@npc$;
	mes "你可以把钻石送给恋人。";
	mes "收到^FF82FF‘17 克拉钻石’^000000的人会珍爱它，";
	mes "并把你珍藏在心中。";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "什么？你没有恋人？";
	mes "你也可以留给自己。";
	mes "^FF82FF‘17 克拉钻石’^000000的迷人光环";
	mes "或许会为你的生活带来爱情。";
	next;
	
	mes .@npc$;
	mes "它被称为钻石中的钻石……";
	mes "这就是^FF82FF‘17 克拉钻石’^000000！";
	next;

	set .@menu$, "^FF82FF'17 Carat Diamond'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Exchange my 17 Carat Diamond for zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "是的！";
		mes "呵呵，你现在看到的^FF82FF‘17 克拉钻石’^000000";
		mes "是品质最高的 17 克拉钻石之一。";
		mes "把它和你从梦罗克挖出的豌豆大小无色石头相比，";
		mes "简直无法相提并论！";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "有人说这就是";
		mes "卡普拉在海洞看到的那颗钻石！";
		next;
		
		if (select("结束对话:多少钱？") == 1) {
			mes .@npc$;
			mes "价格相当便宜！";
			mes "感兴趣的话随时来找我！";
			close;
		}
		
		mes .@npc$;
		mes "当然！最重要的就是价格！";
		mes "正如我之前说的，^FF82FF‘17 克拉钻石’^000000";
		mes "经鉴定确实是 17 克拉钻石。";
		mes "世界上没有多少这样的钻石。";
		next;
		
		mes .@npc$;
		mes "价格只要 5 亿 Zeny！";
		mes "另外还有 1,000,000 Zeny 服务费。";
		mes "和钻石的品质相比，";
		mes "这只是很小的代价。";
		next;
		
		if (select("^828282太贵了。^000000:我买了。") == 1) {
			mes .@npc$;
			mes "说这样的钻石——";
			mes "^FF82FF‘17 克拉钻石’^000000太贵，";
			mes "真让我无话可说。";
			close;
		}
		
		mes .@npc$;
		mes "哦哦哦！从见到你的那一刻起，我就知道你是这颗";
		mes "奇妙钻石的真正主人。";
		mes "很高兴我没有看错！";
		mes "我相信这颗钻石很适合你。";
		mes "你只需完成这份合同。";
		next;
		
		while(1) {
			if (select("^B9062F我会签合同。^000000:^828282算了，我改变主意了。^000000") == 2) {
				mes .@npc$;
				mes "我明白了，真遗憾……";
				close;
			}
			
			mes .@npc$;
			mes "你叫什么名字？";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "我叫" + strcharinfo(0) + "。";
			next;

			mes .@npc$;
			mes "那么，请在这里签名。";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "好的。";
				mes "现在只剩付款了。";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "正在计算价格和服务费……";
					mes "501,000,000 Zeny。";
					mes "金额已确认。";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "你随时可以把钻石换回 Zeny。";
					mes "不过会收取 1,000,000 Zeny 服务费。";
					mes "感谢惠顾。";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "嗯？";
					mes "抱歉，你的资金不足。";
					mes "包括服务费在内，共需要 501,000,000 Zeny。";
					mes "请检查 Zeny 余额后再试。";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "这真的是你的签名吗？";
				mes "要再次签名确认吗？";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "啊，你想把^FF82FF‘17 克拉钻石’^000000换成 Zeny？";
		emotion ET_SURPRISE;
		next;
		if (select("是:否") == 2) {
			mes .@npc$;
			mes "那么，你想要什么呢？……";
			mes "……";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "我明白了。收到你的签名后，";
		mes "你必须交出^FF82FF‘17 克拉钻石’^000000。";
		mes "别忘了还有 1,000,000 Zeny 服务费！";
		next;
		
		mes .@npc$;
		mes "你叫什么名字？";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "我叫" + strcharinfo(0) + "。";
		next;
		
		mes .@npc$;
		mes "那么，请在这里签名。";
		next;
		
		while (1) {
			if (select("签名:不签名") == 2) {
				mes .@npc$;
				mes "没有你的签名就无法完成交易。";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "已收到你的签名。";
				mes "现在我会收取你的^FF82FF‘17 克拉钻石’^000000。";
				mes "之后你会收到^0000FF499,000,000 Zeny^000000。";
				next;
				
				mes .@npc$;
				mes "感谢惠顾。";
				mes "以后需要^FF82FF‘17 克拉钻石’^000000时，";
				mes "随时带着 5 亿 Zeny 再来。";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "这真的是你的签名吗？";
				mes "要再次签名确认吗？";
				next;
			}
		}
	}
}
