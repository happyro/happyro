lou_fild01,240,354,4	script	Mantis Researcher	878,{

	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[Mantis 研究员]";
			mes "你完成了猎杀 50 只螳螂的任务吗？";
			next;
			switch(select("是:否:我想退出")) {
			case 1:
				mes "[Mantis 研究员]";
				mes "嗯，我不认为你已经猎杀了 50 只。";
				close;
			case 2:
				mes "[Mantis 研究员]";
				mes "记住，我需要你帮忙猎杀螳螂，所以去猎杀 50 只吧。";
				close;
			case 3:
				mes "[Mantis 研究员]";
				mes "你确定要停止猎杀吗？";
				mes "你已经取得的进度都会被清除。";
				next;
				switch(select("是:否")) {
				case 1:
					mes "[Mantis 研究员]";
					mes "好吧，如果你改变主意，随时回来。";
					erasequest 60179;
					close;
				case 2:
					mes "[Mantis 研究员]";
					mes "请猎杀 50 只螳螂，让这里成为大家获得经验的地方。";
					close;
				}
			}
		}
		mes "[Mantis 研究员]";
		mes "太棒了，你完成得真快。";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[Mantis 研究员]";
		mes "太好了！";
		mes "你是来帮助我的，对吧？";
		mes "否则我真不知道该拿这里的怪物怎么办！";
		next;
		switch(select("当然:不")) {
		case 1:
			mes "[Mantis 研究员]";
			mes "太好了！我一看到你就知道可以依靠你！";
			mes "我需要你减少当前在这张地图上肆虐的怪物数量！";
			mes "这里螳螂太多了，低等级玩家无法在这里获得良好经验。";
			next;
			mes "[Mantis 研究员]";
			mes "那么，你还愿意帮助我吗？";
			next;
			switch(select("当然:取消")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[Mantis 研究员]";
					mes "谢谢你为猎杀螳螂提供的帮助！";
					mes "现在情况已经稳定，我不再需要你帮忙猎杀它们了。";
					close;
				}
				mes "[Mantis 研究员]";
				mes "猎杀 50 只螳螂，完成后告诉我。我会奖励你的努力。";
				setquest 60179;
				close;
			case 2:
				mes "[Mantis 研究员]";
				mes "改变主意后请回来。";
				close;
			}
		case 2:
			mes "[Mantis 研究员]";
			mes "改变主意后请回来。";
			close;
		}
	}
	mes "[Mantis 研究员]";
	mes "哎呀，哎呀！";
	next;
	mes "[Mantis 研究员]";
	mes "抱歉我现在不能说话，我正痛苦不堪，这些该死的螳螂要杀了我。";
	close;
}
