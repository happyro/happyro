moc_prydb1,39,129,2	script	Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "[盗贼向导]";
				mes "嗯？我认识你吗？你看起来太眼熟了，有点吓人。你没有双胞胎吧？";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[盗贼向导]";
					mes "什么，你想成为盗贼？抱歉，你看起来还需要更多训练。";
					next;
					mes "[盗贼向导]";
					mes "慢慢来，把基础技能都学会，好吗？那么，下次见~！";
					close;
				}
				mes "[盗贼向导]";
				mes "我感觉你经历过一生的战斗，所以现在就晋升你为盗贼。我得满足强者的愿望……";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[盗贼向导]";
				mes "既然成为了盗贼，就以盗贼的身份生活。现在，出发吧！下一个~";
				close;
			}else{
				mes "[盗贼向导]";
				if (Sex == SEX_MALE)
					mes "嘿，伙计。";
				else
					mes "嘿，小可爱~";
				close;
			}
		}else{
			mes "[盗贼向导]";
			if (Sex == SEX_MALE)
				mes "嘿，伙计。";
			else
				mes "嘿，小可爱。";
			mes "……嘿！你看起来太乖了，居然想当盗贼！！快滚，我很忙。下一个！";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[盗贼公会成员]";
		mes "呃……看起来像富裕家庭宠爱的"+(Sex == SEX_MALE?"儿子":"女儿")+"，为什么会来这个肮脏的地方？";
		next;
		mes "[盗贼公会成员]";
		mes "我不负责让你成为盗贼，只负责受理申请，明白吗？";
		mes "想成为盗贼就去问我旁边那个眼尖的家伙。";
		close;
	}
	mes "[盗贼公会成员]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "嘿~遇到麻烦随时来找我，知道吗？";
			close;
		}else{
			mes "搞什么……？";
			if(Sex == SEX_MALE) {
				mes "嗯……你是"+jobname(Class)+"，笨蛋！";
				next;
				mes "[盗贼公会成员]";
				mes "嘿，兄弟。";
			}else{
				next;
				mes "[盗贼公会成员]";
				mes "嘿，女士。";
			}
			mes "你为什么在这里？回你的地方去~回去~~";
			close;
		}
	}
	mes "呃……看起来像富裕家庭宠爱的"+(Sex == SEX_MALE?"儿子":"女儿")+"，为什么会来这个肮脏的地方？";
	next;
	if(select("我想成为盗贼。:没什么。") == 2) {
		mes "[盗贼公会成员]";
		mes "你知道不申请就不能成为盗贼……";
		mes "你在想什么……？";
		close;
	}
	mes "[盗贼公会成员]";
	mes "你就这么骄傲吗？";
	mes "你骄傲地告诉我想当盗贼！为什么不去所有村庄宣传你是盗贼？";
	mes "哈哈哈！挂个横幅写‘我是骄傲的盗贼，专偷别人的东西。’";
	next;
		mes "[盗贼公会成员]";
	mes "你就这么想当盗贼吗？";
	next;
	switch(select("是。:不。:你呢？")) {
	case 1:
		mes "[盗贼公会成员]";
		mes "哦，是吗……嗯……我活着就是为了见识你这样的怪人。";
		next;
		break;
	case 2:
		mes "[盗贼公会成员]";
		mes "那你为什么在这里？你以为成为盗贼这么容易吗？";
		next;
		break;
	case 3:
		mes "[盗贼公会成员]";
		mes "呃……我？我？";
		mes "嗯……我天生适合当盗贼……我不介意这种傻事。";
		next;
		break;
	}
	mes "[盗贼公会成员]";
	mes "总之，在外面的世界千万别说想当盗贼！！";
	next;
	mes "[盗贼公会成员]";
	mes "那么，你要申请成为盗贼吗？";
	next;
	if(select("是的。:不。") == 2) {
		mes "[盗贼公会成员]";
		mes "嗯……做你想做的事吧~走自己的路~";
		close;
	}
	mes "[盗贼公会成员]";
	mes "好，告诉我你的名字。嗯？什么？";
		mes ""+strcharinfo(0)+"。所以你叫"+strcharinfo(0)+"，对吧……？";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[盗贼公会成员]";
		mes "我能看到你成为盗贼的强烈意志……";
		mes "但只有意志无法应对真正的战斗，对吧？";
		mes "去把基础技能至少提升到 9 级。";
		close;
	}
	mes "[盗贼公会成员]";
	mes "嗯……我从侦探社资料里找到了你有趣的犯罪记录。你似乎具备成为优秀盗贼的素质。";
	mes ""+strcharinfo(0)+"。我现在宣布你通过了职业面试！";
	next;
	mes "[盗贼公会成员]";
	mes "我不负责让你成为盗贼，只负责受理申请，明白吗？";
	mes "想成为盗贼就去问我旁边的家伙。";
	set q_job_thief,1;
	close;
}

moc_prydb1,42,133,2	script	Thief Guildsman#thief	118,{
	if(q_job_thief == 1) {
		mes "[盗贼公会成员]";
		mes "好，你一定通过了职业面试吧？";
		mes "很好，我接受你。";
		mes "现在开始公会的转职仪式！";
		next;
		mes "[盗贼公会会长]";
		mes "‘"+strcharinfo(0)+"。’";
		mes "‘月光会为准备好的人开启道路。现在你已完全准备好，我正式允许你成为盗贼公会成员。’";
		callfunc "Job_Change",Job_Thief;
		set q_job_thief,0;
		getitem 13041,1;	// N_Main_Gauche
		next;
		mes "[盗贼公会会长]";
		mes "‘恭喜你成为盗贼。’";
		mes "‘今后请遵守公会规则，成为光荣的成员。’";
		mes "‘如果违反规则给我们带来耻辱，就小心你的后背。’";
		mes "‘总之，我期待你成为伟大的盗贼。’";
		next;
		mes "[布拉德]";
		mes "嘿~呀啊~！恭喜你！我的朋友。";
		mes "我叫‘布拉德’，负责这里的人事工作。";
		mes "现在还说不准，但以后你会有更多机会见到我。";
		next;
		mes "[布拉德]";
		mes "好了，我能为你做的都做了，走你的路吧。我可是很忙的。";
		mes "再见。";
		close;
	}
	if(BaseJob != Job_Novice) {

			mes "现在没有特别活动。走你的路，稍后再来。";
		}else{
			mes "[盗贼公会成员]";
			mes "嘿~嘿~你既不是初心者也不是盗贼！";
			mes "你在这里做什么？这里不欢迎你当家~呜嘿嘿~快出去~";
		}
		close;
	}
	mes "[盗贼公会成员]";
	mes "哦？像你这样的初心者为什么来这里？";
	mes "如果你想成为盗贼，就问我旁边那个坏脾气的女士。";
	close;
}
