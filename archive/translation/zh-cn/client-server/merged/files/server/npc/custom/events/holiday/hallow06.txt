prontera,151,190,5	script	Hallows' Eve Event	807,{
	mes "[Hallows' Eve Event]";
	if(Class == Job_Novice){
		mes "抱歉，您必须先转职才能完成活动。";
		emotion ET_SWEAT, playerattached();
		close;
	}
	if((EVENT_HALWN06&1) && (EVENT_HALWN06&2)){
		mes "抱歉，您已经完成了活动。";
		emotion ET_SORRY;
		close;
	}
	mes "你好，充满活力的人类。";
	next;
	mes "[Hallows' Eve Event]";
	mes "万圣节到了，是时候前往少有人敢去冒险的地方了。";
	next;
	mes "[Hallows' Eve Event]";
	mes "要恶作剧还是要糖果？您喜欢哪一个？";
	next;
	menu "恶作剧",-, "糖果",L_Treat;

	mes "[Hallows' Eve Event]";
	if(EVENT_HALWN06&2){
		mes "抱歉，您已经接受过恶作剧了。现在去领取糖果吧。";
		emotion ET_SORRY;
		close;
	}
	mes "请稍等……";
	next;
	set BaseExp,BaseExp+BaseLevel*1000;
	set EVENT_HALWN06,EVENT_HALWN06|2;

	mes "[Hallows' Eve Event]";
	mes "这是您的糖果。您的基础经验值略有提升。";
	next;
	mes "[Hallows' Eve Event]";
	mes "希望您喜欢这次恶作剧，万圣节快乐！";
	close;

L_Treat:
	mes "[Hallows' Eve Event]";
	if(EVENT_HALWN06&1){
		mes "抱歉，您已经领取过糖果了。现在去接受恶作剧吧。";
		emotion ET_SORRY;
		close;
	}
	if(!(EVENT_HALWN06&4)){
		mes "太好了！一定会很有趣。";
		next;
		mes "[Hallows' Eve Event]";
		mes "我会让您去寻找一些物品。";
		next;
		mes "[Hallows' Eve Event]";
		mes "带回所有需要的物品，我就会为您制作糖果。";
		next;
		mes "[Hallows' Eve Event]";
		mes "这不是普通的可食用糖果，而是可以戴在头上的物品。";
		next;
		mes "[Hallows' Eve Event]";
		mes "不过，制作任何东西都需要材料。";
		mes "清单如下：";
		mes "1顶帽子，";
		mes "30个南瓜，";
		mes "以及20块黑色布片";
		next;
		mes "[Hallows' Eve Event]";
		mes "现在赶快出发吧。";
		set EVENT_HALWN06,EVENT_HALWN06|4;
		close;
	}
	mes "您找齐我要求的物品了吗？需要再看一次清单吗？";
	next;
	menu "找齐了",-, "还没有",L_Dont_Have, "查看清单",L_List;

	mes "[Hallows' Eve Event]";
	if((countitem(2220) < 1) || (countitem(535) < 30) || (countitem(7205) < 20)){
		mes "您的物品数量不足，收集齐后再回来吧。";
		emotion ET_SORRY;
		close;
	}
	delitem 2220,1;
	delitem 535,30;
	delitem 7205,20;
	getitem 5134,1;
	set EVENT_HALWN06,EVENT_HALWN06|1;
	mes "做得好！希望您喜欢这件物品。";
	emotion ET_CONGRATULATION;
	next;
	mes "[Hallows' Eve Event]";
	mes "祝您剩下的万圣节时光愉快。";
	emotion ET_SCISSOR, playerattached();
	close;

L_Dont_Have:
	mes "[Hallows' Eve Event]";
	mes "回去继续寻找吧。要查看清单吗？";
	next;
	menu "否",-, "是",L_List;
	close;

L_List:
	mes "[Hallows' Eve Event]";
	mes "我需要以下物品：";
	mes "1顶帽子，";
	mes "30个南瓜，";
	mes "以及20块黑色布片。";
	close;
}
