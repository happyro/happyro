e_tower,81,105,0	script	Tower Protection Stone	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Endless Tower";

	set .@etower_timer,checkquest(60200,PLAYTIME); // 1 week
	set .@etower_timer2,checkquest(60201,PLAYTIME); // 4 hours

	if (!instance_check_party(.@party_id,2)) {
		mes "请创建或加入人数超过 1 人的队伍后再试。";
		close;
	}
	switch(.@etower_timer) {
	case -1:
		if (is_party_leader() == true) {
			mes "已确认队伍建立。要预约进入无尽塔吗？";
			next;
			switch(select("创建副本 "+.@md_name$+":进入副本:返回艾尔贝塔:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "队伍名称："+ getpartyname(.@party_id);
					mes "队长："+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000- 预约失败！";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000 - 正在尝试预约";
				mes "预约后，请与后方 NPC 交谈并选择‘进入副本’。";
				close;
			case 2:
				callsub L_Enter,0,1;
			case 3:
				mes "我会把你传送到艾尔贝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 4:
				close;
			}
		}
		switch(select("进入 "+.@md_name$+":返回艾尔贝塔:取消")) {
		case 1:
			callsub L_Enter,1,1;
		case 2:
			mes "我会把你传送到艾尔贝塔。";
			close2;
			warp "alberta",223,36;
			end;
		case 3:
			end;
		}
	case 0:
	case 1:
		if (.@etower_timer2 < 2 && getcharid(1) == etower_partyid) {
			mes "如果副本已经创建，就可以进入。";
			next;
			switch(select("进入 "+.@md_name$+":返回艾尔贝塔:取消")) {
			case 1:
				callsub L_Enter,0,0;
			case 2:
				mes "我会把你传送到艾尔贝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 3:
				close;
			}
		} else {
			set .@dun_lim_time,etower_timer+604800; // 1 week
			// set .@dun_lim_time2,etower_timer+14400; // 4 hours
			set .@dun_cur_time,gettimetick(2);
			set .@dun_ent_t,(.@dun_lim_time - .@dun_cur_time);
			set .@dun_h,(.@dun_ent_t / 3600);
			set .@dun_m,(.@dun_ent_t - (.@dun_h * 3600)) / 60;
			set .@dun_s,.@dun_ent_t - ((.@dun_h * 3600) + (.@dun_m * 60));

			mes "受高塔余波影响，现在无法进入副本。距离下次进入还剩 " + .@dun_h + " 小时 " + .@dun_m + " 分钟 " + .@dun_s + " 秒。";
			next;
			mes "这里很危险。我把你传送到艾尔贝塔。";
			close2;
			warp "alberta",223,36;
			end;
		}
	case 2:
		set etower_timer,0;
		erasequest 60200;
		erasequest 60201;
		mes "^0000ff与无尽塔相关的记录和后续效果已移除。你可以再次生成并进入无尽塔。^000000";
		close;
	}

L_Enter:
	switch(instance_enter("Endless Tower")) {
	case IE_OTHER:
		mes "发生未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "纪念副本无尽塔不存在。";
		mes "队长尚未创建副本。";
		close;
	case IE_NOMEMBER:
		mes "建立队伍后即可进入副本。";
		close;
	case IE_OK:
		mapannounce "e_tower", strcharinfo(0) +" 所在的队伍正在进入副本无尽塔。",bc_map,"0x00ff99",FW_NORMAL,12;
		if (getarg(1)) {
			set etower_timer,gettimetick(2);
			set etower_partyid, getcharid(1);
			setquest 60200;
			setquest 60201;
		}
		//warp "1@tower",52,354;
		if (getarg(0) == 0) close;
		else end;
	}
}
