// Guild Flags (Outside Castle)
//============================================================
-	script	Scarlet Palace#1::OutsideFlagsPA2	722,{

	set .@GID, GetCastleData("payg_cas02",1);

	if (.@GID == 0) {
		mes "[ 神圣卢恩·米德加尔特王国法令 ]";
		mes " ";
		mes "1. 遵循神圣卢恩·米德加尔特王国的法令，";
		mes "我们宣布：";
		mes "这座城堡目前没有正式主人。";
		mes " ";
		mes "2. 对于能够";
		mes "克服所有考验";
		mes "并摧毁 Emperium 的人，";
		mes "国王将授予其";
		mes "这座城堡的所有权。";
		close;
	}
	if (getcharid(2) == .@GID) {
	mes "[ 回荡的声音 ]";
	mes "勇敢的人啊……";
	mes "你想回到属于你的荣耀之地吗？";
		next;
		if(select("返回公会城堡。:退出。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("payg_cas02",1)) {
					warp "payg_cas02",278,251;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神圣卢恩·米德加尔特王国法令 ]";
	mes " ";
	mes "1. 遵循神圣卢恩·米德加尔特王国的法令，";
	mes "我们确认此地属于";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公会的私人领地。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公会的会长是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如果有人对此有异议，";
	mes "就用手中的钢刃证明你的力量与荣誉。";
	close;

OnRecvCastlePy02:
	FlagEmblem GetCastleData("payg_cas02",1);
	end;
}
