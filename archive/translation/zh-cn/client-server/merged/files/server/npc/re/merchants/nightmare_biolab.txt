		sc_start SC_BLEEDING,5000,0;
	}

		showscript "这里好冷……";

		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 6) {
		showscript "留下来……陪着我……";
		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 7) {
		showscript "你不该来这里……";
		sc_start SC_STUN,5000,0;
	}
	else if (.@r == 8) {
		showscript "你好……有人在吗……？";
		sc_start SC_BLIND,5000,0;
	}
	else if (.@r == 9) {
		showscript "我好害怕……好害怕……";
		sc_start SC_CONFUSION,10000,0;
	}
	else if (.@r == 10) {
		showscript "亲爱的奥丁……";
		sc_start SC_ASSUMPTIO,100000,5;

	}

	}

		showscript "对不起……我现在就离开……谢谢……";
