job3_rune01,88,62,5	script	Dragon Breeder	105,{
	mes "[飞龙饲养员]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "欢迎。要租一条龙吗？";
		next;
		switch(select("是:否")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[飞龙饲养员]";
				mes "请先学习骑龙技能。";
				close;
			}
			else if (checkdragon()) {
				mes "[飞龙饲养员]";
				mes "你已经有一条龙了。";
				close;
			}
			else if (ismounting()) {
				mes "[飞龙饲养员]";
				mes "请先解除现金坐骑。";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[飞龙饲养员]";
			mes "我明白了。祝你今天愉快。";
			close;
		}
	}
	mes "你在这里做什么？";
	mes "只有符文骑士可以租龙。";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[魔导机甲装甲师]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "欢迎，机械工匠。";
		mes "要租手推车，还是";
		mes "骑魔导机甲？";
		next;
		switch(select("租手推车:骑魔导机甲:购买紧急魔导机甲:升级冷却装置:取消")) {
		case 1:
			if (checkcart()) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你已经";
				mes "有手推车了。";
				close;
			}
			setcart;
			mes "[魔导机甲装甲师]";
			mes "给你！";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你已经在骑";
				mes "魔导机甲了。";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[魔导机甲装甲师]";
				mes "请先学习获得魔导机甲驾照的技能。";
				close;
			}
			else if(ismounting()) {
				mes "[魔导机甲装甲师]";
				mes "请先解除现金坐骑。";
				close;
			}
			setmadogear;
			mes "[魔导机甲装甲师]";
			mes "玩得开心，欢迎再来！";
			close;
		case 3:
			mes "[魔导机甲装甲师]";
			mes "紧急魔导机甲在紧急情况下非常有用，售价 1,000,000 Zeny。";
			next;
			if (select("购买:取消") == 2) {
				mes "[魔导机甲装甲师]";
				mes "我明白了。改变主意后";
				mes "随时可以来找我。";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你已经 have an Emergency Mado Gear.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你没有足够的 Zeny 购买紧急魔导机甲。";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[魔导机甲装甲师]";
			mes "给你！";
			close;
		case 4:
			mes "[魔导机甲装甲师]";
			mes "你想升级哪个装置？";
			next;
			if (select("冷却装置:高品质冷却器") == 1) {
				mes "[魔导机甲装甲师]";
				mes "将冷却装置升级为高品质冷却器需要 1 个冷却装置和 2,000,000 Zeny。";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[魔导机甲装甲师]";
				mes "将高品质冷却器升级为特殊冷却器需要 1 个高品质冷却器和 4,000,000 Zeny。";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("升级:取消") == 2) {
				mes "[魔导机甲装甲师]";
				mes "我明白了。改变主意后";
				mes "随时可以来找我。";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你没有" + getitemname(.@itemid) + "。";
				close;
			}
			if (Zeny < .@cost) {
				mes "[魔导机甲装甲师]";
				mes "抱歉，你没有足够的 Zeny 升级装置。";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[魔导机甲装甲师]";
			mes "给你！这是你的" + getitemname(.@itemid) + "。";
			close;
		case 5:
			close;
		}
	}
	mes "我能帮你什么？";
	mes "魔导机甲只提供给机械工匠。";
	close;
}
