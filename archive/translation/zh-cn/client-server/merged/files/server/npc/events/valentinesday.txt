// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[Stephen]";
	mes "猜猜我有什么？";
	mes "米德加尔特不易找到的美味……";
	next;
	mes "[Stephen]";
	mes "巧克力！";
	mes "没错，你不喜欢巧克力吗……我喜欢。";
	mes "你很幸运，我只卖 5,000 Zeny 一块！";
	next;
	if(select("我想要巧克力！:不用了，谢谢。") == 2) {
		mes "[Stephen]";
		mes "你不想要巧克力？";
		mes "我告诉你，你会后悔的！";
		mes "现在最好买一些……以后再也遇不到这样的巧克力了！";
		mes "考虑一下，改天再来。";
		close; }
	mes "[Stephen]";
	mes "哈！";
	mes "我就知道！";
	mes "但一次最多只能卖 5 块……如果你真的需要更多……";
	mes "可以再回来。";
	mes "那么你要多少？";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[Stephen]";
	if (.@i > 5)  {
		mes "抱歉，我不能给你那么多。";
		close; }
	if (Zeny < .@i*5000) {
		mes "抱歉，看来你买不起这些。";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "给你！";
	mes "可以送给别人当礼物，也可以自己享用！";
	mes "嗯……甜美的巧克力……";
	mes "随时欢迎来找我……！";
	close;
}
