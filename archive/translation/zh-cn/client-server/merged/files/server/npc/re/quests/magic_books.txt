function	script	Magic_Book_Essence	{
	if (MaxWeight - Weight < 1000) {
		mes "请减轻背包重量。";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "你使用了 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 制作^999900"+ getarg(3) +"精华^000000。";
			next;
			mes "^008800除非发生问题，否则"+getarg(3)+"精华将永久有效。^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888使用 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 精炼精华失败。^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "你已经在这个区域制作过"+getarg(3)+"精华了。";
		close;
	}
	mes "已经有人使用过这个区域了。";
	close;
}
