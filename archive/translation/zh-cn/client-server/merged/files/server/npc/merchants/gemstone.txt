payon,173,238,5	script	Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF等一下！你现在携带的物品太多了。请先把一些物品存入卡普拉仓库后再来。^000000";
		close;
	}
	mes "[Jade]";
	mes "带给我两颗";
	mes "相同颜色的宝石，";
	mes "我就能把它们变成";
	mes "不同颜色的宝石。";
	mes "另一种颜色的宝石。";
	next;
	switch(select("把蓝色宝石换成红色！:把红色宝石换成黄色！:把黄色宝石换成蓝色！")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[Jade]";
		mes "哈……！";
		mes "你是在开玩笑吧？";
		mes "如果你至少不给我";
		mes "2 颗"+ getitemname(.@item_req) +"，";
		mes "我就无法为你提供服务。";
		mes "我就无法提供这项服务！";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[Jade]";
		mes "我可以使用你目前拥有的";
		mes getitemname(.@item_req) +"，";
		mes "总共制作 " + .@gems + " 颗";
		mes getitemname(.@item_id) +"。";
		mes "你想怎么做？";
		next;
		switch(select("尽可能多地给我。:我想设定数量。:我不换了。")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[Jade]";
			mes "那么你想要多少？";
			mes "你可以输入的最大数量是 100。";
			mes "请输入一个合适的数量。";
			mes "请告诉我你需要的数量。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Jade]";
					mes "一颗都不要？";
					mes "我猜你改变主意了……";
					mes "那就下次再来吧。";
					close;
				}
				else if (.@input > 100) {
					mes "[Jade]";
					mes "呃……";
					mes "我让你输入不超过 100 的数量，";
					mes "记得吗……？";
					mes "请重新输入。";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[Jade]";
					mes "呃……";
					mes "你没有那么多宝石";
					mes "可以用来兑换……";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[Jade]";
			mes "当然，没问题。";
			mes "随时欢迎回来。";
			close;
		}
		mes "[Jade]";
		mes "给你。";
		mes "随时欢迎";
		mes "回来。";
		mes "嗯，你为什么这样看我？";
		mes "我的脸上有什么吗？";
		close;
	}
}
