S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[Madeleine Chu]";
	mes "你想购买多少个"+ (.@item_id == 12125 ? "户外":"室内");
	mes "烹饪套装？";
	mes "如果";
	mes "想取消，请";
	mes "输入数字0。";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[Madeleine Chu]";
			mes "改变主意了？";
			mes "以后如果需要购买";
			mes ""+ getitemname(.@item_id);
			mes "套装，随时回来找我";
			mes "就好。";
			close;
		}
		else if (.@sell > 100) {
			mes "[Madeleine Chu]";
			mes "哦，抱歉，";
			mes "为了安全，我一次最多只卖";
			mes "100个"+ getitemname(.@item_id);
			mes "。";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[Madeleine Chu]";
		mes "哦，抱歉，你";
		mes "买不起这么多";
		mes getitemname(.@item_id);
		mes "购买套装前请确认";
		mes "Zeny充足～";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[Madeleine Chu]";
		mes "抱歉，你的";
		mes "背包没有足够空间放下这么多";
		mes "物品";
		mes getitemname(.@item_id) +"……";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[Madeleine Chu]";
	mes "给你～";
	mes "祝你的";
	mes "烹饪训练顺利！";
	return;
}
