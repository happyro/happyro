valkyrie,53,50,3	script	Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[生化学家]";
		mes "啊，你是来";
		mes "找回记忆";
		mes "失去的记忆吗？是的，你";
		mes "一定是为了曾经属于你的";
		mes "生命秘密";
		mes "而来……";
		next;
		if (select("是:否") == 1) {
			mes "[生化学家]";
			mes "闭上眼睛";
			mes "放松心神。";
			mes "我们将回到你的";
			mes "过去，重新收集";
			mes "失落记忆的碎片。";
			mes "。";
			next;
			mes "[生化学家]";
			mes "睁开眼睛后，";
			mes "你会清楚记得";
			mes "生命的秘密，也会记得";
			mes "使用这些秘密时";
			mes "肩负的责任，";
			mes "并将它们用于正途……";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[生化学家]";
			mes "睁开眼睛……";
			mes "现在你已经";
			mes "记起如何";
			mes "创造人工生命，我只希望你";
			mes "尊重你所有的";
			mes "创造物。";
			close;
		}
		mes "[生化学家]";
		mes "如果你想";
		mes "找回失去的";
		mes "记忆，请";
		mes "回来找我。";
		mes "创造生命的秘密";
		mes "绝非小事……";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[生化学家]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[生化学家]";
		mes "像我这样的人";
		mes "出现在这里很奇怪。";
		mes "但即使精通";
		mes "科学之道的人";
		mes "也能成为英雄。";
		next;
		mes "[生化学家]";
		mes "在这种情况下，";
		mes "重要的不一定是";
		mes "我使用的手段，而是";
		mes "我为之奋斗的目标，";
		mes "它们让我赢得";
		mes "英灵殿的一席之地……";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[生化学家]";
		mes "是的……";
		mes "是时候了。";
		mes "米德加尔特需要更多";
		mes "像你这样的天才。";
		next;
		mes "[生化学家]";
		mes "你想成为";
		mes "生化学家吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[生化学家]";
			mes "准备好后，";
			mes "随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[生化学家]";
			mes "你仍然可以学习更多技能。请在回来前使用";
			mes "所有剩余技能点。";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[生化学家]";
		mes "恭喜！";
		mes "作为生化学家，";
		mes "希望你将渊博的知识";
		mes "用于正确的目的。";
		mes "不要辜负这份力量。";
		close;
	}
	else {
		mes "[生化学家]";
		mes "欢迎来到";
		mes "英灵殿，";
		mes "荣誉之厅。";
		next;
		mes "[生化学家]";
		mes "请在这里";
		mes "安心休息。";
		mes "英灵殿欢迎每一位来客。";
		mes "向勇士致敬！";
		close;
	}
}
