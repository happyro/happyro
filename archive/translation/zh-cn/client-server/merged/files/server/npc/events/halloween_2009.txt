prontera,152,192,5	script	Pumpkin Hat Researcher	712,{
	if(BaseLevel < 45) {
		mes "[Pumpkin Hat Researcher]";
		mes "走开，我不需要小孩子。走开！我不和初心者说话。";
		next;
		mes "[Pumpkin Hat Researcher]";
		mes "等你达到能与更强怪物战斗的等级后再回来。";
		close;
	}
	mes "[Pumpkin Hat Researcher]";
	mes "你喜欢南瓜派吗？";
	next;
	mes "[Pumpkin Hat Researcher]";
	mes "我是南瓜帽研究者，要不要听听我的故事？";
	next;
	while(1) {
		switch(select("听故事。:询问南瓜帽。:领取南瓜帽。:结束对话。")) {
			case 1:
				mes "[Pumpkin Hat Researcher]";
				mes "我一直在研究升级版南瓜帽。";
				mes "我发现升级过程非常简单。";
				next;
				mes "[Pumpkin Hat Researcher]";
				mes "过程相当简单。";
				mes "带给我^4a4aff20个南瓜^000000，我就能给你演示。";
				mes "这个提议很诱人吧？";
				next;
				break;
			case 2:
				mes "[Pumpkin Hat Researcher]";
				mes "这顶升级版南瓜帽非常强大！";
				next;
				mes "[Pumpkin Hat Researcher]";
				mes "它能用凝聚的能量制作南瓜派，为佩戴者恢复大量HP和SP。";
				next;
				mes "[Pumpkin Hat Researcher]";
				mes "你只需要^4a4aff20个南瓜^000000。";
				next;
				break;
			case 3:
				mes "[Pumpkin Hat Researcher]";
				mes "想得到南瓜派吗？好，让我数数你带来的南瓜。";
				next;
				if(countitem(1062) < 20) {
					mes "[Pumpkin Hat Researcher]";
					mes "我需要^4a4aff20个南瓜^000000。";
					mes "我不是炼金术士或魔法师，不能凭空变出东西。";
					next;
					mes "[Pumpkin Hat Researcher]";
					mes "好，去猎杀怪物后再回来。";
					mes "我会在这里待一阵子，不用着急。";
					next;
					break;
				}else{
					mes "[Pumpkin Hat Researcher]";
					mes "希望它对你有用。别忘了战斗时戴上它，才能得到南瓜派。";
					delitem 1062,20;	// Jack o' Pumpkin
					getitem 5668,1;		// Weird Pumpkin Hat
					next;
					mes "[Pumpkin Hat Researcher]";
					mes "我想我该回去继续研究了。";
					close;
				}
			case 4:
				mes "[Pumpkin Hat Researcher]";
				mes "再见，下次再见。";
				mes "祝你一切顺利……";
				close;
		}
	}
}

-	script	Halloween Wizard#iRO09::09HallowWiz	737,{
	mes "[Halloween Wizard]";
	mes "...";
	mes "你想捉弄某个人吗？";
	next;
	while(1) {
		switch(select("什么捉弄？:当然:不要。")) {
			case 1:
				mes "[Halloween Wizard]";
				mes "只要一些材料，我就能在世界各地召唤怪物。";
				next;
				mes "[Halloween Wizard]";
				mes "听起来很有趣吧？";
				next;
				mes "[Halloween Wizard]";
				mes "带给我布料、南瓜、旧布料或碎南瓜，我就能召唤怪物。";
				next;
				break;
			case 2:
				mes "[Halloween Wizard]";
				mes "你想在哪座城镇捉弄人？";
				next;
				getmapxy(.@mapname$,.@mapx,.@mapy,BL_PC,""+strcharinfo(0)+"");
				if (.@mapname$ == "prontera") {
					switch(select("吉芬:斐扬:艾尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,3;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "payon") {
					switch(select("普隆德拉:吉芬:艾尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "geffen") {
					switch(select("普隆德拉:斐扬:艾尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "alberta") {
					switch(select("普隆德拉:吉芬:斐扬:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "aldebaran") {
					switch(select("普隆德拉:吉芬:斐扬:艾尔贝塔")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,4;
							break;
					}
				}
				setarray .@HallowTowns$[1],"prontera","payon","geffen","alberta","aldebaran";
				mes "[Halloween Wizard]";
				mes "好，进入下一步。";
				next;
				mes "[Halloween Wizard]";
				mes "你想使用多少布料或南瓜？不要超过100个，这是我能使用的最大数量。";
				next;
				input .@input;
				if (.@input == 0) {
					mes "[Halloween Wizard]";
					mes "你还没想好。";
					mes "没关系。";
					mes "想好后告诉我。";
					next;
					break;
				}
				else if (.@input > 100) {
					mes "[Halloween Wizard]";
					mes "我说过必须在1到100之间！";
					mes "你没注意听！";
					next;
					break;
				}
				else {
					set .@fabric,countitem(1059);
					set .@jack,countitem(1062);
					set .@worn,countitem(6299);
					set .@crushed,countitem(6298);
					set .@whispers,0;
					set .@darklords,0;

					set .@total,.@fabric + .@jack + .@worn + .@crushed;

					if(.@total < .@input) {
						mes "[Halloween Wizard]";
						mes "重新数数你拥有的物品，把总数告诉我。";
						mes "呵呵呵呵呵……";
						next;
						break;
					}

					if(.@fabric > 0) {
						if(.@fabric >= .@input) {
							delitem 1059,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 1059,.@fabric;
							set .@input,.@input - .@fabric;
							set .@whispers,.@whispers+.@fabric;
						}
					}
					if(.@worn > 0 && .@input != 0) {
						if(.@worn >= .@input) {
							delitem 6299,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 6299,.@worn;
							set .@input,.@input - .@worn;
							set .@whispers,.@whispers+.@worn;
						}
					}
					if(.@jack > 0 && .@input != 0) {
						if(.@jack >= .@input) {
							delitem 1062,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 1062,.@jack;
							set .@input,.@input - .@jack;
							set .@darklords,.@darklords+.@jack;
						}
					}
					if(.@crushed > 0 && .@input != 0) {
						if(.@crushed >= .@input) {
							delitem 6298,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 6298,.@crushed;
							set .@input,.@input - .@crushed;
							set .@darklords,.@darklords+.@crushed;
						}
					}
					if (.@input > 0) {
						mes "出问题了。";
						close;
					}
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Whisper",3014,.@whispers;
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Dark Lord",3015,.@darklords;
					mes "[Halloween Wizard]";
					mes "这就是你想要的。";
					mes "想象一下其他村庄的人会怎么想？";
					mes "哇哈哈哈哈";
					close;
				}
			case 3:
				mes "[Halloween Wizard]";
				mes "改变主意后就回来吧……";
				mes "我会在这里待一阵子……";
				mes "咯咯咯……";
				close;
		}
	}
}
