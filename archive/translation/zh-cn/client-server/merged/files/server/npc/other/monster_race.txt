p_track01,58,0,0	script	Race Progress Timer	-1,{
OnEnable:
	initnpctimer;
	enablenpc "Race Progress Timer";
	for ( .@c = 1; .@c < 7; .@c++ ) {
		.@line = rand(1,70);
		enablenpc "starting#"+.@c;
		enablenpc "Tire"+.@c+"#1";
		if ( .@line <= 10 ) {
			.@tired = rand(50,60);
			enablenpc "Luk"+.@c+"#5";
			enablenpc "Luk"+.@c+"#6";
		} else if ( .@line <= 30 ) {
			.@tired = rand(40,60);
			enablenpc "Luk"+.@c+"#5";
			if (.@tired >= 50) enablenpc "Tire"+.@c+"#2";
		} else if ( .@line <= 40 ) {
			.@tired = rand(30,50);
			enablenpc "Luk"+.@c+"#1";
			enablenpc "Tire"+.@c+"#2";
			if ( .@tired < 40 ) enablenpc "Tire"+.@c+"#3";
		} else if (.@line <= 50) {
			.@tired = rand(20,40);
			enablenpc "Luk"+.@c+"#1";
			enablenpc "Luk"+.@c+"#2";
			enablenpc "Tire"+.@c+"#2";
			enablenpc "Tire"+.@c+"#3";
			if (.@tired < 30) enablenpc "Tire"+.@c+"#4";
		} else if (.@line <= 60) {
			.@tired = rand(10,30);
			for ( .@i = 1; .@i <= 3; .@i++ )
				enablenpc "Luk"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 4; .@i++ )
				enablenpc "Tire"+.@c+"#"+.@i;
			if ( .@tired < 20) enablenpc "Tire"+.@c+"#5";
		} else if (.@line <= 70) {
			.@tired = rand(0,20);
			for ( .@i = 1; .@i <= 4; .@i++ )
				enablenpc "Luk"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 5; .@i++ )
				enablenpc "Tire"+.@c+"#"+.@i;
			if (.@tired < 10) enablenpc "Tire"+.@c+"#6";
		}
		setd "$@mr_1_luk"+.@c, .@line;
		setd "$@mr_1_tire"+.@c, .@tired;
	}
	end;
OnTimer1000:
	enablenpc "售票员#single";
	end;
OnTimer7000:

	end;
OnTimer10000:
	mapannounce "p_track01","如有疑问，欢迎随时向服务台咨询。",bc_map,"0xffb6c1";
	end;
OnTimer120000:
	mapannounce "p_track01","单人怪物赛将在 3 分钟后开始。",bc_map,"0xffb6c1";
	end;
OnTimer123000:
	mapannounce "p_track01","如需下注，请向售票员咨询。",bc_map,"0xffb6c1";
	end;
OnTimer240000:
	mapannounce "p_track01","单人怪物赛即将开始。",bc_map,"0xffb6c1";

OnTimer243000:
	mapannounce "p_track01","如需下注，请向售票员咨询。",bc_map,"0xffb6c1";

OnTimer300000:

	.@select = select("单人赛入口—开启:双人赛入口—开启:单人赛入口—关闭:双人赛入口—关闭");
	mes "[Monster Race Manager]";
	mes "请输入";
	mes "密码。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,2000);
	if (.@i == -2) {
		mes "[Monster Race Manager]";
		mes "错误。";
	} else if (.@i == -1) {
		mes "[Monster Race Manager]";
		mes "密码错误。";
	} else if (.@i == 1) {
		mes "[Monster Race Manager]";
		switch(.@select) {
		case 1:
			mes "单人怪物赛跑";
			mes "入口 NPC 已开启。";
			donpcevent "Eckar Ellebird#single::OnEnable";
			close;
		case 2:
			mes "双人怪物赛跑";
			mes "入口 NPC 已开启。";
			donpcevent "Eckar Erenes#double::OnEnable";
			close;
		case 3:
			mes "单人怪物赛跑";
			mes "入口 NPC 已关闭。";
			donpcevent "Eckar Ellebird#single::OnDisable";
			close;
		case 4:
			mes "双人怪物赛跑";
			mes "入口 NPC 已关闭。";
			donpcevent "Eckar Erenes#double::OnDisable";
			close;
		}
	}
	close;
}
