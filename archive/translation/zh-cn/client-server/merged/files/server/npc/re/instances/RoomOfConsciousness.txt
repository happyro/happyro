1@mir,103,80,3	script	Fenrir#boss1a	4_F_FENRIR,2,2,{
	end;
OnEvent:
	'fenrir_name$ = instance_npcname( strnpcinfo(0) );
	'iris_name$ = instance_npcname( "Iris#" + strnpcinfo(2) );
	enablenpc 'fenrir_name$;
	enablenpc 'iris_name$;
	end;

OnTouchNPC:
	if ('touch_mob == 0) {
		npctalk "Fenrir：干得好！接下来交给你了，Iris！", 'fenrir_name$;
		'touch_mob = 1;
		initnpctimer;
	}
	end;
OnTimer2000:
		npctalk "Iris：好，交给我吧！我，Iris，想要的是……！", 'iris_name$;
	specialeffect EF_BEGINSPELL,AREA, 'iris_name$;
	end;
OnTimer4000:
		npctalk "Iris：Mabi 护符！去吧！！", 'iris_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "啊啊啊——！！";
	else
		unittalk 'boss_id, "Bijou：什么……！！";
	end;
OnTimer5000:
		npctalk "Iris：成功了！", 'iris_name$;
	end;
OnTimer7000:
		npctalk "Fenrir：那么轮到我了。~精神十足地大喊~", 'fenrir_name$;
	specialeffect EF_BEGINSPELL,AREA, 'fenrir_name$;
	specialeffect EF_TETRACASTING,AREA, 'fenrir_name$;
	progressbar_npc "000000",3;
	end;
OnTimer10000:
		npctalk "Fenrir：接招！！", 'fenrir_name$;
	specialeffect EF_SUI_EXPLOSION,AREA, 'fenrir_name$;
	specialeffect EF_LORD,AREA, 'fenrir_name$;
	specialeffect EF_FLAMELAUNCHER,AREA, 'fenrir_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "~惨叫~";
	else
		unittalk 'boss_id, "Bijou：~呻吟~比我想象的更痛……！";
	getunitdata 'boss_id, .@data;
	if (.@data[UMOB_HP] > 0) {
		.@damage = rand(700,1300) * 1000;
		if (.@damage >= .@data[UMOB_HP])
			.@mob_hp = 0;
		else
			.@mob_hp = .@data[UMOB_HP] - .@damage;
		setunitdata 'boss_id, UMOB_HP, .@mob_hp;
	}
	end;
OnTimer13000:
		npctalk "Fenrir：虽然不算完美，但我觉得我们造成了相当大的伤害！", 'fenrir_name$;
	end;
OnTimer14500:
		npctalk "Fenrir：我消耗了太多魔力。我要稍微休息一下。接下来拜托你了！", 'fenrir_name$;
	end;
OnTimer16500:
	donpcevent instance_npcname("eq#mir3") + "::OnStart";
OnStop:
	stopnpctimer;
	disablenpc();
	disablenpc instance_npcname( "Iris#" + strnpcinfo(2) );
	'touch_mob = 0;
	'random_letter$ = "";
	end;
}
