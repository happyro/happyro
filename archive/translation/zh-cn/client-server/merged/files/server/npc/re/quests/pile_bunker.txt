yuno,179,174,4	script	Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[Gomer]";
		mes "你携带的东西太多了。放下一些行李后再来找我。";
		close;
	}
	if (countitem(1549)) {
		mes "[Gomer]";
		mes "那把重型尖刺铁锤用得怎么样？";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[Gomer]";
		mes "我是退休的机匠，恐怕没什么能帮你的。";
		close;
	}
	mes "[Gomer]";
	mes "怎么样？驾驶魔导机甲是什么感觉？";
	mes "没能找到自己道路的人，是不允许驾驶它的。";
	next;
	mes "[Gomer]";
	mes "我？我很久以前就退休了。因为遭遇事故……受了重伤，再也不能驾驶。我对此很难过。";
	next;
	mes "[Gomer]";
	mes "所以我决定留在这里为魔导机甲锻造齿轮。";
	mes "嗯？我不是说讨厌这份工作。";
	next;
	mes "[Gomer]";
	mes "被迫做不喜欢的事，没有比这更痛苦的了。";
	next;
	mes "[Gomer]";
	mes "也有人甘愿只当陪衬就很开心，比如成为供别人倚靠的墙。";
	next;
	mes "[Gomer]";
	mes "这让我感觉很好，而且我正要制作新的齿轮。想帮我吗？";
	next;
	switch(select("我没那么感兴趣。:你要制作什么？:材料从哪里得到？:我带来了材料。")) {
	case 1:
		mes "[Gomer]";
		mes "那我也无能为力了。";
		mes "下次见！";
		close;
		case 2:
	mes "[Gomer]";
		mes "嗯，一根能一击砸碎对手的巨大金属棍！";
		next;
		mes "[Gomer]";
		mes "它看起来不华丽，但我设计了弹簧装置，能以巨大力量发动攻击。";
		next;
		mes "[Gomer]";
		mes "它是隐藏式的，找到机会时一击就能让对手陷入重伤！";
		mes "我迫不及待想看它实战！";
		next;
		mes "[Gomer]";
		mes "只要你想象这种武器有多厉害，我就能把它实现。";
		next;
		mes "[Gomer]";
		mes "带给我 200 个钢铁、30 个柔软管和 1 把布罗卡。我可以用这些材料制作刚才描述的武器。";
		close;
		case 3:
	mes "[Gomer]";
		mes "你可以从邪恶箱等许多怪物身上获得钢铁。";
		next;
		mes "[Gomer]";
		mes "柔软管更稀有，由艾因布洛克和拉赫附近的金属波利掉落。";
		next;
		mes "[Gomer]";
		mes "最后，布罗卡很难找到，但你应该能想办法。哈哈，去拿来吧！";
		close;
		case 4:
	if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
		mes "[Gomer]";
			mes "你没有带齐我要求的所有材料。";
			next;
			mes "[Gomer]";
			mes "我有灵感时就会放弃正在制作的东西，尝试制作新的东西。";
			mes "你最好快点！";
			close;
			} else {
		mes "[Gomer]";
			mes "哦，我一直在等你。你没按时来，我还以为你忘了。请在那里稍等一会儿。";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[Gomer]";
			mes "完成了。你觉得怎么样？还不错吧？虽然是我亲手做的，但确实做得相当好。";
			next;
			mes "[Gomer]";
			mes "重量、体积、颜色和光泽的组合。真让我心动！";
			next;
			mes "[Gomer]";
			mes "...........";
			next;
			mes "[Gomer]";
			mes "嗯，再稍加改进就能量产交付。我肯定还能做出更多，所以这把就送给你。";
			next;
			mes "[Gomer]";
			mes "哦，结果我不需要那么多钢铁。所以拿回大部分吧。";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
			}
		}
	}
